```python
"""Structured logging and run.json manifests."""

from __future__ import annotations

import json
import logging
import sys
from datetime import datetime, timezone
from pathlib import Path
from typing import Any


def setup_logging(level: int = logging.INFO) -> logging.Logger:
    """Configure root logger for console output."""
    logging.basicConfig(
        level=level,
        format="%(asctime)s | %(levelname)s | %(name)s | %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S",
        stream=sys.stdout,
        force=True,
    )
    return logging.getLogger("slm-calib")


def utc_now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


def write_run_json(
    path: str | Path,
    *,
    phase: int,
    phase_name: str,
    status: str,
    payload: dict[str, Any] | None = None,
    seed: int | None = None,
    config_hash: str | None = None,
    git_commit: str | None = None,
) -> dict[str, Any]:
    """Write a run.json manifest and return the dict."""
    run: dict[str, Any] = {
        "phase": phase,
        "phase_name": phase_name,
        "status": status,
        "timestamp_utc": utc_now_iso(),
        "seed": seed,
        "config_hash": config_hash,
        "git_commit": git_commit,
    }
    if payload:
        run.update(payload)

    out = Path(path)
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(run, indent=2, default=str), encoding="utf-8")
    return run
```

```python
from __future__ import annotations

import os
import random
import subprocess
from typing import Any


def set_seed(seed: int = 42, deterministic: bool = True) -> None:
    """Seed python, numpy, and torch RNGs."""
    os.environ["PYTHONHASHSEED"] = str(seed)
    random.seed(seed)
    try:
        import numpy as np

        np.random.seed(seed)
    except ImportError:
        pass
    try:
        import torch

        torch.manual_seed(seed)
        if torch.cuda.is_available():
            torch.cuda.manual_seed_all(seed)
        if deterministic:
            torch.backends.cudnn.deterministic = True
            torch.backends.cudnn.benchmark = False
    except ImportError:
        pass


def get_git_commit(repo_root: str | None = None) -> str | None:
    """Return current git commit hash, or None if unavailable."""
    try:
        result = subprocess.run(
            ["git", "rev-parse", "HEAD"],
            capture_output=True,
            text=True,
            cwd=repo_root,
            check=False,
        )
        if result.returncode == 0:
            return result.stdout.strip()
    except (OSError, FileNotFoundError):
        pass
    return None


def collect_env_snapshot() -> dict[str, Any]:
    """Collect lightweight environment info for run manifests."""
    info: dict[str, Any] = {
        "python_version": __import__("platform").python_version(),
        "platform": __import__("platform").platform(),
        "executable": __import__("sys").executable,
    }
    try:
        import torch

        info["torch_version"] = torch.__version__
        info["cuda_available"] = torch.cuda.is_available()
        if torch.cuda.is_available():
            info["cuda_version"] = torch.version.cuda
            info["gpu_name"] = torch.cuda.get_device_name(0)
    except ImportError:
        info["torch_version"] = "NOT INSTALLED"
```

```python
from typing import Any

from omegaconf import DictConfig, OmegaConf


def load_config(path: str | Path, overrides: list[str] | None = None) -> DictConfig:
    """Load a YAML config file with optional dotlist overrides."""
    cfg = OmegaConf.load(str(path))
    if overrides:
        cfg = OmegaConf.merge(cfg, OmegaConf.from_dotlist(overrides))
    return cfg


def config_hash(cfg: DictConfig | dict[str, Any]) -> str:
    """Stable hash of a config object for run manifests."""
    import hashlib
    import json

    container = OmegaConf.to_container(cfg, resolve=True) if isinstance(cfg, DictConfig) else cfg
    payload = json.dumps(container, sort_keys=True, default=str)
    return hashlib.sha256(payload.encode("utf-8")).hexdigest()[:16]
```

### `src/data/splits.py`

```python
"""Leave-one-context-out (LOCO) splits, hashing, and integrity checks."""

from __future__ import annotations

import hashlib
import json
from pathlib import Path
from typing import Any

import pandas as pd
from sklearn.model_selection import train_test_split


def _class_priors(series: pd.Series) -> dict[str, float]:
    vc = series.value_counts(normalize=True)
    return {str(k): float(v) for k, v in vc.items()}


def create_loco_splits(
    df: pd.DataFrame,
    domain_col: str = "domain",
    *,
    seed: int = 42,
    min_domain_size: int = 200,
    id_col: str = "example_id",
    train_frac: float = 0.7,
) -> dict[str, Any]:
    """Create LOCO split definitions.

    For each domain D with enough samples:
      - source = all rows where domain != D
      - target = rows where domain == D  (ood_test)
      - source is stratified into train / id_val / id_test

    Returns dict keyed by ``loco_<D>`` with example-ID lists and class priors.
    """
    if domain_col not in df.columns:
        raise ValueError(f"domain_col {domain_col!r} not in dataframe")
    if id_col not in df.columns:
        df = df.copy()
        df[id_col] = [f"row_{i}" for i in range(len(df))]

    domain_sizes = df[domain_col].astype(str).value_counts()
    domains = sorted([d for d, n in domain_sizes.items() if n >= min_domain_size])
    if len(domains) < 2:
        raise ValueError(
            f"Need >=2 domains with size>={min_domain_size}; got {dict(domain_sizes)}"
        )

    splits: dict[str, Any] = {}
    for held_out in domains:
        source = df[df[domain_col].astype(str) != held_out].copy()
        target = df[df[domain_col].astype(str) == held_out].copy()
        if len(source) < 10 or len(target) < 5:
            continue

        strat = source["label"] if source["label"].nunique() > 1 else None
        try:
            train, temp = train_test_split(
                source,
                test_size=1.0 - train_frac,
                stratify=strat,
                random_state=seed,
            )
        except ValueError:
            train, temp = train_test_split(
                source, test_size=1.0 - train_frac, random_state=seed
            )
        strat_temp = temp["label"] if temp["label"].nunique() > 1 else None
        val_frac_of_temp = 1.0 / 3.0  # temp is 30% → val 10%, test 20%
        try:
            id_val, id_test = train_test_split(
                temp,
                test_size=1.0 - val_frac_of_temp,
                stratify=strat_temp,
                random_state=seed,
            )
        except ValueError:
            id_val, id_test = train_test_split(
                temp, test_size=1.0 - val_frac_of_temp, random_state=seed
            )

        train_ids = train[id_col].astype(str).tolist()
        id_val_ids = id_val[id_col].astype(str).tolist()
        id_test_ids = id_test[id_col].astype(str).tolist()
        ood_test_ids = target[id_col].astype(str).tolist()

        splits[f"loco_{held_out}"] = {
            "target_domain": held_out,
            "source_domains": [d for d in domains if d != held_out],
            "train_ids": train_ids,
            "id_val_ids": id_val_ids,
            "id_test_ids": id_test_ids,
            "ood_test_ids": ood_test_ids,
            "class_priors": {
                "train": _class_priors(train["label"]),
                "id_val": _class_priors(id_val["label"]),
                "id_test": _class_priors(id_test["label"]),
                "ood_test": _class_priors(target["label"]),
            },
            "sizes": {
                "train": len(train_ids),
                "id_val": len(id_val_ids),
                "id_test": len(id_test_ids),
                "ood_test": len(ood_test_ids),
            },
            "seed": seed,
            "domain_col": domain_col,
        }
    if not splits:
        raise ValueError("No LOCO splits were produced")
    return splits


def freeze_and_hash_splits(splits: dict[str, Any], output_path: str | Path) -> str:
    """Write splits JSON deterministically and return SHA-256 of content."""
    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    # sort_keys for byte-stable hashing regardless of insertion order
    content = json.dumps(splits, sort_keys=True, indent=2, ensure_ascii=False)
    sha = hashlib.sha256(content.encode("utf-8")).hexdigest()
    output_path.write_text(content, encoding="utf-8")
    sha_path = Path(str(output_path) + ".sha256")
    sha_path.write_text(sha + "\n", encoding="utf-8")
    print(f"Splits saved to {output_path} | SHA-256: {sha}")
    return sha


def load_splits(path: str | Path) -> tuple[dict[str, Any], str]:
    path = Path(path)
    content = path.read_text(encoding="utf-8")
    sha = hashlib.sha256(content.encode("utf-8")).hexdigest()
    return json.loads(content), sha


def verify_split_integrity(splits: dict[str, Any]) -> dict[str, Any]:
    """T1.5 / T1.7 checks: disjoint IDs, coverage, no target leakage."""
    report: dict[str, Any] = {"ok": True, "splits": {}}
    for name, spec in splits.items():
        train = set(spec["train_ids"])
        id_val = set(spec["id_val_ids"])
        id_test = set(spec["id_test_ids"])
        ood = set(spec["ood_test_ids"])

        issues: list[str] = []
        if train & id_val:
            issues.append("train∩id_val")
        if train & id_test:
            issues.append("train∩id_test")
        if train & ood:
            issues.append("train∩ood_test")
        if id_val & id_test:
            issues.append("id_val∩id_test")
        if id_val & ood:
            issues.append("id_val∩ood_test")
        if id_test & ood:
            issues.append("id_test∩ood_test")
        source_union = train | id_val | id_test
        # source union should be disjoint from ood (already checked pairwise)
        entry = {
            "issues": issues,
            "n_train": len(train),
            "n_id_val": len(id_val),
            "n_id_test": len(id_test),
            "n_ood_test": len(ood),
            "source_covers_only_source": not (source_union & ood),
        }
        if issues:
            report["ok"] = False
        report["splits"][name] = entry
    return report
```

```python
"""Task-performance metrics for binary hate-speech classification."""

from __future__ import annotations

from typing import Any

import numpy as np
from sklearn.metrics import (
    accuracy_score,
    average_precision_score,
    f1_score,
    matthews_corrcoef,
    precision_score,
    recall_score,
    roc_auc_score,
)

_NEG_INF = float("nan")


def _safe(fn, *args, **kwargs) -> float:
    """Run a sklearn metric, mapping degenerate inputs (single class) to NaN."""
    try:
        return float(fn(*args, **kwargs))
    except ValueError:
        return _NEG_INF


def compute_all_task_metrics(
    y_true: Any, y_pred: Any, y_prob: Any
) -> dict[str, float]:
    """Compute all task-performance metrics for a binary task."""
    yt = np.asarray(y_true).reshape(-1)
    yp = np.asarray(y_pred).reshape(-1)
    yprob = np.asarray(y_prob, dtype=np.float64).reshape(-1)
    if not (yt.shape[0] == yp.shape[0] == yprob.shape[0]):
        raise ValueError("y_true, y_pred and y_prob must have equal length")
    if yt.shape[0] == 0:
        raise ValueError("empty input")

    single_class = np.unique(yt).size < 2
    return {
        "n": int(yt.shape[0]),
        "accuracy": _safe(accuracy_score, yt, yp),
        "balanced_accuracy": _safe(
            lambda a, b: float(
                0.5 * ((np.asarray(b) == 1)[np.asarray(a) == 0].mean())
                + ((np.asarray(b) == 0)[np.asarray(a) == 1].mean())
            )
            if np.unique(yt).size == 2
            else float("nan"),
            yt,
            yp,
        ),
        "macro_f1": _safe(f1_score, yt, yp, average="macro", zero_division=0),
        "weighted_f1": _safe(f1_score, yt, yp, average="weighted", zero_division=0),
        "hate_f1": _safe(f1_score, yt, yp, pos_label=1, zero_division=0),
        "nonhate_f1": _safe(f1_score, yt, yp, pos_label=0, zero_division=0),
        "macro_precision": _safe(precision_score, yt, yp, average="macro", zero_division=0),
        "macro_recall": _safe(recall_score, yt, yp, average="macro", zero_division=0),
        "hate_precision": _safe(precision_score, yt, yp, pos_label=1, zero_division=0),
        "hate_recall": _safe(recall_score, yt, yp, pos_label=1, zero_division=0),
        "auroc": _NEG_INF if single_class else _safe(roc_auc_score, yt, yprob),
        "average_precision": _NEG_INF
        if single_class
        else _safe(average_precision_score, yt, yprob),
        "mcc": _safe(matthews_corrcoef, yt, yp),
        "positive_rate": float(yt.mean()),
        "pred_positive_rate": float(np.mean(np.asarray(yp) == 1)),
    }


def agreement_rate(pred_a: Any, pred_b: Any) -> float:
    """Fraction of examples where two prediction vectors agree."""
    a = np.asarray(pred_a).reshape(-1)
    b = np.asarray(pred_b).reshape(-1)
    if a.shape[0] != b.shape[0]:
        raise ValueError("length mismatch")
    return float((a == b).mean())


def bootstrap_accuracy_delta(
    y_true: Any, pred_a: Any, pred_b: Any, n_boot: int = 1000, seed: int = 42
) -> dict[str, float]:
    """Paired bootstrap of ``acc(a) - acc(b)`` (used for quantization deltas)."""
    yt = np.asarray(y_true).reshape(-1)
    a = (np.asarray(pred_a).reshape(-1) == yt).astype(np.float64)
    b = (np.asarray(pred_b).reshape(-1) == yt).astype(np.float64)
    n = yt.shape[0]
    if n == 0:
        raise ValueError("empty input")
    rng = np.random.default_rng(seed)
    deltas = np.empty(n_boot, dtype=np.float64)
    for i in range(n_boot):
        idx = rng.integers(0, n, size=n)
        deltas[i] = a[idx].mean() - b[idx].mean()
    lo, hi = np.percentile(deltas, [2.5, 97.5])
    return {
        "delta": float(a.mean() - b.mean()),
        "ci_low": float(lo),
        "ci_high": float(hi),
        "p_delta_le_0": float((deltas <= 0).mean()),
        "n_boot": int(n_boot),
    }
```

### `src/data/dedup.py`

```python
"""Exact and near-duplicate removal for Phase 1 data prep."""

from __future__ import annotations

import hashlib
from typing import Any

import pandas as pd

try:
    from datasketch import MinHash, MinHashLSH
except ImportError:  # pragma: no cover
    MinHash = None  # type: ignore
    MinHashLSH = None  # type: ignore


def normalize_text_series(series: pd.Series) -> pd.Series:
    return (
        series.astype(str)
        .str.lower()
        .str.strip()
        .str.replace(r"\s+", " ", regex=True)
    )


def exact_dedup(df: pd.DataFrame, text_col: str = "text") -> pd.DataFrame:
    """Remove exact duplicates after normalization."""
    df = df.copy()
    df["_text_norm"] = normalize_text_series(df[text_col])
    before = len(df)
    # Keep first occurrence; stable by original order
    df = df.drop_duplicates(subset="_text_norm", keep="first")
    after = len(df)
    print(f"Exact dedup: {before} -> {after} ({before - after} removed)")
    return df.drop(columns=["_text_norm"]).reset_index(drop=True)


def _minhash_for_text(text: str, num_perm: int) -> MinHash:
    m = MinHash(num_perm=num_perm)
    # Character 3-grams are robust for morphologically rich Bengali.
    t = text if text else ""
    if len(t) < 3:
        grams = [t] if t else [""]
    else:
        grams = [t[i : i + 3] for i in range(len(t) - 2)]
    for g in grams:
        m.update(g.encode("utf-8"))
    return m


def near_dedup(
    df: pd.DataFrame,
    text_col: str = "text",
    threshold: float = 0.8,
    num_perm: int = 128,
    id_col: str | None = None,
) -> pd.DataFrame:
    """Remove near-duplicates via MinHash LSH (Jaccard threshold).

    Dedup is global across the dataframe (all domains / splits present).
    The first surviving example of each near-dup cluster is kept.
    """
    if MinHashLSH is None:
        raise ImportError("datasketch is required for near_dedup")
    df = df.copy().reset_index(drop=True)
    if id_col is None:
        id_col = "example_id" if "example_id" in df.columns else "_row_id"
        if id_col == "_row_id":
            df[id_col] = [str(i) for i in range(len(df))]

    lsh = MinHashLSH(threshold=threshold, num_perm=num_perm)
    texts = df[text_col].astype(str).tolist()
    keys = df[id_col].astype(str).tolist()
    keep_mask = [True] * len(df)

    for i, (key, text) in enumerate(zip(keys, texts)):
        mh = _minhash_for_text(text, num_perm=num_perm)
        try:
            # Query for already-indexed near-duplicates before inserting.
            neighbors = lsh.query(mh)
            if neighbors:
                keep_mask[i] = False
                continue
            lsh.insert(key, mh)
        except ValueError:
            # Same key already present → treat as duplicate.
            keep_mask[i] = False

    removed = int(sum(1 for k in keep_mask if not k))
    print(f"Near dedup (Jaccard>={threshold}): removed {removed} of {len(df)}")
    out = df.loc[keep_mask].copy()
    if id_col == "_row_id":
        out = out.drop(columns=["_row_id"])
    return out.reset_index(drop=True)


def dedup_pipeline(
    df: pd.DataFrame,
    *,
    text_col: str = "text",
    near_threshold: float = 0.8,
    num_perm: int = 128,
    run_near: bool = True,
) -> tuple[pd.DataFrame, dict[str, Any]]:
    """Exact + optional near dedup; returns cleaned frame and stats."""
    stats: dict[str, Any] = {"n_raw": int(len(df))}
    df1 = exact_dedup(df, text_col=text_col)
    stats["n_exact"] = int(len(df1))
    if run_near:
        df2 = near_dedup(
            df1, text_col=text_col, threshold=near_threshold, num_perm=num_perm
        )
    else:
        df2 = df1
    stats["n_final"] = int(len(df2))
    stats["removed_exact"] = stats["n_raw"] - stats["n_exact"]
    stats["removed_near"] = stats["n_exact"] - stats["n_final"]
    return df2, stats
```

### `src/metrics/selective.py`

```python
"""Selective-prediction metrics: risk-coverage, AURC, E-AURC, coverage@risk.

Confidence is the top-class probability ``max(p, 1-p)``; examples are ranked by
descending confidence and the risk-coverage curve reports the error rate over the
accepted prefix.
"""

from __future__ import annotations

from typing import Any

import numpy as np

_EPS = 1e-15


def _trapz(y: np.ndarray, x: np.ndarray) -> float:
    """Trapezoidal integration (numpy renamed trapz -> trapezoid in 1.26/2.x)."""
    fn = getattr(np, "trapezoid", None) or np.trapz
    return float(fn(y, x))


def _prepare(y_true: Any, y_prob: Any) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    yt = np.asarray(y_true, dtype=np.float64).reshape(-1)
    yp = np.asarray(y_prob, dtype=np.float64).reshape(-1)
    if yt.shape[0] != yp.shape[0]:
        raise ValueError(f"length mismatch: y_true={yt.shape[0]} y_prob={yp.shape[0]}")
    if yt.shape[0] == 0:
        raise ValueError("empty input")
    yp = np.clip(yp, 0.0, 1.0)
    confidence = np.maximum(yp, 1.0 - yp)
    prediction = (yp >= 0.5).astype(np.int64)
    errors = (prediction != yt).astype(np.float64)
    return yt, yp, confidence, errors


def _integrate(risks: np.ndarray, coverages: np.ndarray, method: str) -> float:
    """Area under the discrete risk-coverage curve.

    ``riemann`` (default) is the standard definition used in the selective
    prediction literature: the mean of the per-coverage risks.  ``trapezoid``
    is the variant sketched in the project plan.
    """
    if method == "riemann":
        return float(risks.mean())
    if method == "trapezoid":
        return _trapz(risks, coverages)
    raise ValueError(f"Unknown method: {method}")


def risk_coverage_curve(y_true: Any, y_prob: Any) -> tuple[np.ndarray, np.ndarray]:
    """Return ``(coverages, risks)`` with one point per accepted example."""
    _, _, confidence, errors = _prepare(y_true, y_prob)
    n = confidence.shape[0]
    order = np.argsort(-confidence, kind="stable")
    sorted_errors = errors[order]
    coverages = np.arange(1, n + 1) / n
    risks = np.cumsum(sorted_errors) / np.arange(1, n + 1)
    return coverages, risks


def aurc(y_true: Any, y_prob: Any, method: str = "riemann") -> float:
    """Area Under the Risk-Coverage curve (lower is better)."""
    coverages, risks = risk_coverage_curve(y_true, y_prob)
    return _integrate(risks, coverages, method)


def optimal_aurc(n: int, n_errors: int, method: str = "riemann") -> float:
    """AURC of a perfect ranker: every error sits at the *lowest* confidence.

    With all ``n_errors`` errors deferred to the final ranks, the cumulative
    error count at rank ``k`` is ``max(0, k - (n - n_errors))``, so
    ``risk(k) = max(0, k - (n - n_errors)) / k``.
    """
    n = int(n)
    n_errors = int(n_errors)
    if n <= 0:
        raise ValueError("n must be positive")
    n_errors = int(np.clip(n_errors, 0, n))
    ks = np.arange(1, n + 1)
    risks = np.maximum(0.0, ks - (n - n_errors)) / ks
    coverages = ks / n
    return _integrate(risks, coverages, method)


def e_aurc(y_true: Any, y_prob: Any, method: str = "riemann") -> float:
    """Excess AURC = AURC(model) - AURC(optimal ranker).

    Isolates *ranking* quality from raw accuracy (Geifman & El-Yaniv, 2017).
    """
    _, _, _, errors = _prepare(y_true, y_prob)
    n = errors.shape[0]
    return float(aurc(y_true, y_prob, method) - optimal_aurc(n, int(errors.sum()), method))


def selective_accuracy_at_coverage(y_true: Any, y_prob: Any, coverage: float) -> float:
    """Accuracy on the most-confident ``coverage`` fraction of examples."""
    if not 0.0 < coverage <= 1.0:
        raise ValueError(f"coverage must be in (0, 1], got {coverage}")
    yt, yp, confidence, _ = _prepare(y_true, y_prob)
    n = yt.shape[0]
    n_select = int(round(n * coverage))
    if n_select == 0:
        return float("nan")
    top = np.argsort(-confidence, kind="stable")[:n_select]
    return float(((yp[top] >= 0.5).astype(np.int64) == yt[top]).mean())


def coverage_at_risk(y_true: Any, y_prob: Any, max_risk: float = 0.05) -> float:
    """Largest coverage whose running risk stays <= ``max_risk``."""
    coverages, risks = risk_coverage_curve(y_true, y_prob)
    ok = risks <= max_risk
    if not ok.any():
        return 0.0
    return float(np.max(coverages[ok]))


def failure_prediction_auroc(y_true: Any, y_prob: Any) -> float:
    """AUROC of using ``1 - confidence`` to predict an error (higher is better)."""
    from sklearn.metrics import roc_auc_score

    yt, _, confidence, errors = _prepare(y_true, y_prob)
    n = yt.shape[0]
    if errors.sum() == 0 or errors.sum() == n:
        return float("nan")
    return float(roc_auc_score(errors.astype(int), 1.0 - confidence))


def compute_all_selective_metrics(y_true: Any, y_prob: Any) -> dict[str, float]:
    """All selective-prediction metrics in one call."""
    out: dict[str, float] = {
        "aurc": aurc(y_true, y_prob),
        "e_aurc": e_aurc(y_true, y_prob),
        "coverage_at_10pct_risk": coverage_at_risk(y_true, y_prob, 0.10),
        "coverage_at_5pct_risk": coverage_at_risk(y_true, y_prob, 0.05),
        "coverage_at_2pct_risk": coverage_at_risk(y_true, y_prob, 0.02),
        "failure_prediction_auroc": failure_prediction_auroc(y_true, y_prob),
    }
    for cov in (0.1, 0.3, 0.5, 0.7, 0.9, 1.0):
        out[f"selective_accuracy_at_{int(cov * 100)}pct"] = selective_accuracy_at_coverage(
            y_true, y_prob, cov
        )
    return out
```

```python
"""Domain-shift magnitude metrics (embedding + lexical)."""

from __future__ import annotations

import re
from typing import Any, Sequence

import numpy as np

_WS_RE = re.compile(r"\s+")


def _as_list(texts: Sequence[str]) -> list[str]:
    return ["" if t is None else str(t) for t in texts]


def _tokenize_words(text: str) -> list[str]:
    return [t for t in _WS_RE.split(text.lower().strip()) if t]


def _char_ngrams(text: str, n: int = 3) -> set[str]:
    t = _WS_RE.sub(" ", text.lower().strip())
    if len(t) < n:
        return {t} if t else set()
    return {t[i : i + n] for i in range(len(t) - n + 1)}


def compute_mmd(
    x: np.ndarray,
    y: np.ndarray,
    kernel: str = "rbf",
    gamma: float | None = None,
) -> float:
    """Unbiased-ish MMD^2 with RBF kernel (median heuristic if gamma is None)."""
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    if x.ndim == 1:
        x = x[:, None]
    if y.ndim == 1:
        y = y[:, None]
    if len(x) == 0 or len(y) == 0:
        return float("nan")

    def _sq_dists(a: np.ndarray, b: np.ndarray) -> np.ndarray:
        a2 = np.sum(a**2, axis=1)[:, None]
        b2 = np.sum(b**2, axis=1)[None, :]
        return np.maximum(a2 + b2 - 2.0 * (a @ b.T), 0.0)

    d_xx = _sq_dists(x, x)
    d_yy = _sq_dists(y, y)
    d_xy = _sq_dists(x, y)

    if gamma is None:
        all_d = np.concatenate([d_xx.ravel(), d_yy.ravel(), d_xy.ravel()])
        all_d = all_d[all_d > 0]
        med = float(np.median(all_d)) if len(all_d) else 1.0
        gamma = 1.0 / med if med > 0 else 1.0

    if kernel != "rbf":
        raise ValueError(f"Unsupported kernel: {kernel}")

    k_xx = np.exp(-gamma * d_xx)
    k_yy = np.exp(-gamma * d_yy)
    k_xy = np.exp(-gamma * d_xy)

    n = len(x)
    m = len(y)
    # Exclude diagonal for unbiased estimate when possible
    if n > 1:
        term_x = (k_xx.sum() - np.trace(k_xx)) / (n * (n - 1))
    else:
        term_x = k_xx.mean()
    if m > 1:
        term_y = (k_yy.sum() - np.trace(k_yy)) / (m * (m - 1))
    else:
        term_y = k_yy.mean()
    term_xy = k_xy.mean()
    mmd2 = term_x + term_y - 2.0 * term_xy
    return float(max(mmd2, 0.0))


def centroid_cosine_dist(emb_a: np.ndarray, emb_b: np.ndarray) -> float:
    a = np.asarray(emb_a, dtype=np.float64).mean(axis=0)
    b = np.asarray(emb_b, dtype=np.float64).mean(axis=0)
    na = np.linalg.norm(a)
    nb = np.linalg.norm(b)
    if na == 0 or nb == 0:
        return float("nan")
    return float(1.0 - np.dot(a, b) / (na * nb))


def compute_vocab_overlap(source_texts: Sequence[str], target_texts: Sequence[str]) -> float:
    """Jaccard overlap of word-type sets."""
    src = set()
    tgt = set()
    for t in _as_list(source_texts):
        src.update(_tokenize_words(t))
    for t in _as_list(target_texts):
        tgt.update(_tokenize_words(t))
    if not src and not tgt:
        return 1.0
    if not src or not tgt:
        return 0.0
    return float(len(src & tgt) / len(src | tgt))


def compute_oov_rate(
    source_texts: Sequence[str],
    target_texts: Sequence[str],
    *,
    unit: str = "word",
) -> float:
    """Fraction of target tokens/types unseen in the source vocabulary."""
    src = set()
    for t in _as_list(source_texts):
        if unit == "word":
            src.update(_tokenize_words(t))
        else:
            src.update(_char_ngrams(t))
    if not src:
        return 1.0
    total = 0
    oov = 0
    for t in _as_list(target_texts):
        if unit == "word":
            toks = _tokenize_words(t)
        else:
            toks = list(_char_ngrams(t))
        total += len(toks)
        oov += sum(1 for tok in toks if tok not in src)
    if total == 0:
        return 1.0
    return float(oov / total)


def embed_texts(
    texts: Sequence[str],
    model_name: str,
    *,
    batch_size: int = 64,
    max_samples: int | None = 2000,
) -> np.ndarray:
    """Encode texts with a sentence-transformers model (CPU/GPU auto)."""
    from sentence_transformers import SentenceTransformer

    texts = _as_list(texts)
    if max_samples is not None and len(texts) > max_samples:
        idx = np.linspace(0, len(texts) - 1, num=max_samples).astype(int)
        texts = [texts[i] for i in idx]
    model = SentenceTransformer(model_name)
    emb = model.encode(
        texts,
        batch_size=batch_size,
        show_progress_bar=False,
        convert_to_numpy=True,
        normalize_embeddings=False,
    )
    return np.asarray(emb, dtype=np.float64)


def compute_shift_magnitude(
    source_texts: Sequence[str],
    target_texts: Sequence[str],
    *,
    model_name: str = "paraphrase-multilingual-MiniLM-L12-v2",
    embeddings: tuple[np.ndarray, np.ndarray] | None = None,
    max_samples: int | None = 2000,
) -> dict[str, Any]:
    """Compute MMD, centroid cosine distance, vocab overlap, and OOV rate."""
    source_texts = _as_list(source_texts)
    target_texts = _as_list(target_texts)
    if embeddings is None:
        src_emb = embed_texts(source_texts, model_name, max_samples=max_samples)
        tgt_emb = embed_texts(target_texts, model_name, max_samples=max_samples)
    else:
        src_emb, tgt_emb = embeddings

    return {
        "mmd": compute_mmd(src_emb, tgt_emb),
        "centroid_cosine_dist": centroid_cosine_dist(src_emb, tgt_emb),
        "vocab_overlap": compute_vocab_overlap(source_texts, target_texts),
        "oov_rate_word": compute_oov_rate(source_texts, target_texts, unit="word"),
        "oov_rate_char3": compute_oov_rate(source_texts, target_texts, unit="char"),
        "n_source": len(source_texts),
        "n_target": len(target_texts),
        "embedding_model": model_name,
    }


def shift_rows_from_splits(
    df,
    splits: dict[str, Any],
    *,
    text_col: str = "text",
    id_col: str = "example_id",
    model_name: str = "paraphrase-multilingual-MiniLM-L12-v2",
    max_samples_per_side: int = 1500,
) -> list[dict[str, Any]]:
    """For each LOCO split, measure source(train) → ood_test shift magnitude."""
    id_to_text = dict(zip(df[id_col].astype(str), df[text_col].astype(str)))
    rows: list[dict[str, Any]] = []
    for name, spec in splits.items():
        src_ids = spec["train_ids"]
        tgt_ids = spec["ood_test_ids"]
        src_texts = [id_to_text[i] for i in src_ids if i in id_to_text]
        tgt_texts = [id_to_text[i] for i in tgt_ids if i in id_to_text]
        metrics = compute_shift_magnitude(
            src_texts,
            tgt_texts,
            model_name=model_name,
            max_samples=max_samples_per_side,
        )
        priors = spec.get("class_priors", {})
        prior_src = priors.get("train", {})
        prior_tgt = priors.get("ood_test", {})
        metrics.update(
            {
                "split": name,
                "source_domain": "source",
                "target_domain": spec.get("target_domain"),
                "prior_train_hate": prior_src.get("1", prior_src.get(1)),
                "prior_ood_hate": prior_tgt.get("1", prior_tgt.get(1)),
            }
        )
        rows.append(metrics)
    return rows
```

```python
"""Tokenizer fertility measurement (tokens per word / character)."""

from __future__ import annotations

from typing import Any, Sequence

import numpy as np


def measure_tokenizer_fertility(
    model_id: str,
    texts: Sequence[str],
    language: str,
    *,
    max_samples: int | None = None,
) -> dict[str, Any]:
    """Measure tokens-per-word and tokens-per-character for a tokenizer.

    Higher fertility means the model's representation is weaker / more
    fragmented for that language.
    """
    from transformers import AutoTokenizer

    texts = ["" if t is None else str(t) for t in texts]
    if max_samples is not None and len(texts) > max_samples:
        idx = np.linspace(0, len(texts) - 1, num=max_samples).astype(int)
        texts = [texts[i] for i in idx]

    tokenizer = AutoTokenizer.from_pretrained(model_id, trust_remote_code=False)
    total_tokens = 0
    total_words = 0
    total_chars = 0
    oov_tokens = 0

    unk_id = tokenizer.unk_token_id
    for text in texts:
        try:
            tokens = tokenizer.encode(text, add_special_tokens=False)
        except Exception:
            tokens = tokenizer(text, add_special_tokens=False)["input_ids"]
        if unk_id is not None:
            oov_tokens += sum(1 for t in tokens if t == unk_id)
        total_tokens += len(tokens)
        words = text.split()
        total_words += len(words)
        total_chars += len(text)

    return {
        "model": model_id,
        "language": language,
        "tokens_per_word": (total_tokens / total_words) if total_words else float("nan"),
        "tokens_per_char": (total_tokens / total_chars) if total_chars else float("nan"),
        "sample_size": len(texts),
        "total_tokens": total_tokens,
        "unk_token_rate": (oov_tokens / total_tokens) if total_tokens else float("nan"),
        "status": "ok",
    }


def measure_all_fertility(
    model_ids: Sequence[str],
    bn_texts: Sequence[str],
    en_texts: Sequence[str],
    *,
    n_samples: int = 1000,
    skip_on_error: bool = True,
) -> list[dict[str, Any]]:
    """Run fertility measurement for each model on Bengali + English samples."""
    rows: list[dict[str, Any]] = []
    for model_id in model_ids:
        for lang, texts in (("bn", bn_texts), ("en", en_texts)):
            try:
                row = measure_tokenizer_fertility(
                    model_id, texts, lang, max_samples=n_samples
                )
            except Exception as e:
                if not skip_on_error:
                    raise
                row = {
                    "model": model_id,
                    "language": lang,
                    "status": f"error: {type(e).__name__}: {e}",
                    "sample_size": 0,
                }
            rows.append(row)
            print(f"fertility {model_id} [{lang}]: {row.get('status')} "
                  f"tpw={row.get('tokens_per_word')} n={row.get('sample_size')}")
    return rows
```

### `src/metrics/calibration.py`

```python
"""Calibration metrics: ECE, Brier (+ Murphy decomposition), NLL, confidence gap.

All functions take binary labels ``y_true`` (0/1) and ``y_prob`` = P(class 1).
Conventions follow Guo et al. (2017) and Murphy (1973); netcal is used as an
independent cross-check in tests.
"""

from __future__ import annotations

from typing import Any

import numpy as np

_EPS = 1e-15


def _as_float_array(x: Any) -> np.ndarray:
    return np.asarray(x, dtype=np.float64).reshape(-1)


def _validate(y_true: Any, y_prob: Any) -> tuple[np.ndarray, np.ndarray]:
    yt = _as_float_array(y_true)
    yp = _as_float_array(y_prob)
    if yt.shape[0] != yp.shape[0]:
        raise ValueError(f"length mismatch: y_true={yt.shape[0]} y_prob={yp.shape[0]}")
    if yt.shape[0] == 0:
        raise ValueError("empty input")
    return yt, yp


def bin_indices(y_prob: np.ndarray, n_bins: int = 15, strategy: str = "equal_mass") -> list[np.ndarray]:
    """Return a list of index arrays, one per bin.

    ``equal_mass``  -> quantile bins over the sorted probabilities (equal count).
    ``equal_width`` -> uniform bins on [0, 1].
    """
    yp = _as_float_array(y_prob)
    if n_bins < 1:
        raise ValueError(f"n_bins must be >= 1, got {n_bins}")
    if strategy == "equal_mass":
        order = np.argsort(yp, kind="stable")
        return [np.asarray(chunk, dtype=int) for chunk in np.array_split(order, n_bins)]
    if strategy == "equal_width":
        edges = np.linspace(0.0, 1.0, n_bins + 1)
        yp_clip = np.clip(yp, 0.0, 1.0)
        out = []
        for i in range(n_bins):
            lo, hi = edges[i], edges[i + 1]
            mask = (yp_clip >= lo) & (yp_clip < hi) if i < n_bins - 1 else (yp_clip >= lo) & (yp_clip <= hi)
            out.append(np.where(mask)[0])
        return [np.asarray(c, dtype=int) for c in out]
    raise ValueError(f"Unknown strategy: {strategy}")


def expected_calibration_error(
    y_true: Any,
    y_prob: Any,
    n_bins: int = 15,
    strategy: str = "equal_mass",
) -> float:
    """Expected Calibration Error (lower is better).

    ECE = sum_b (|B_b| / N) * |acc(B_b) - conf(B_b)|
    """
    yt, yp = _validate(y_true, y_prob)
    ece = 0.0
    for idx in bin_indices(yp, n_bins=n_bins, strategy=strategy):
        if idx.size == 0:
            continue
        ece += idx.size / yt.shape[0] * abs(yt[idx].mean() - yp[idx].mean())
    return float(ece)


def maximum_calibration_error(
    y_true: Any, y_prob: Any, n_bins: int = 15, strategy: str = "equal_mass"
) -> float:
    """MCE: the largest single-bin |acc - conf| gap."""
    yt, yp = _validate(y_true, y_prob)
    worst = 0.0
    for idx in bin_indices(yp, n_bins=n_bins, strategy=strategy):
        if idx.size == 0:
            continue
        worst = max(worst, abs(yt[idx].mean() - yp[idx].mean()))
    return float(worst)


def brier_score(y_true: Any, y_prob: Any) -> float:
    """Brier score = mean squared error of the positive-class probability."""
    yt, yp = _validate(y_true, y_prob)
    return float(np.mean((np.clip(yp, 0.0, 1.0) - yt) ** 2))


def brier_decomposition(
    y_true: Any, y_prob: Any, n_bins: int = 15, strategy: str = "equal_width"
) -> dict[str, float]:
    """Exact binned Brier decomposition.

    For bins ``b`` with mass ``m_b``, mean forecast ``f_b`` and observed
    frequency ``o_b`` (Murphy 1973; Kumar et al. 2019):

        Brier = reliability - resolution + uncertainty
                + within_bin_variance + within_bin_covariance

    * ``reliability          = sum_b m_b (o_b - f_b)^2``
    * ``resolution            = sum_b m_b (o_b - o_bar)^2``
    * ``uncertainty           = o_bar (1 - o_bar)``
    * ``within_bin_variance   = sum_b m_b Var(p | b)``
    * ``within_bin_covariance = -2 sum_b m_b Cov(y, p | b)``

    The classic three-term Murphy form drops the last two terms, so it equals the
    Brier score only when each bin holds a single forecast value.  All terms are
    reported here so ``brier == brier_score`` exactly; ``brier_murphy_3term``
    keeps the textbook value for comparability with published results.
    """
    yt, yp = _validate(y_true, y_prob)
    yp = np.clip(yp, 0.0, 1.0)
    n = yt.shape[0]
    base_rate = float(yt.mean())
    uncertainty = float(base_rate * (1.0 - base_rate))

    reliability = 0.0
    resolution = 0.0
    within_bin_variance = 0.0
    within_bin_covariance = 0.0
    for idx in bin_indices(yp, n_bins=n_bins, strategy=strategy):
        if idx.size == 0:
            continue
        m_b = idx.size / n
        conf_b = float(yp[idx].mean())
        acc_b = float(yt[idx].mean())
        reliability += m_b * (acc_b - conf_b) ** 2
        resolution += m_b * (acc_b - base_rate) ** 2
        within_bin_variance += m_b * float(yp[idx].var())
        centred = (yt[idx] - acc_b) * (yp[idx] - conf_b)
        within_bin_covariance += -2.0 * m_b * float(centred.mean())

    three_term = reliability - resolution + uncertainty
    return {
        "reliability": float(reliability),
        "resolution": float(resolution),
        "uncertainty": uncertainty,
        "within_bin_variance": float(within_bin_variance),
        "within_bin_covariance": float(within_bin_covariance),
        "brier_murphy_3term": float(three_term),
        "brier": float(three_term + within_bin_variance + within_bin_covariance),
        "brier_direct": brier_score(yt, yp),
    }


def negative_log_likelihood(y_true: Any, y_prob: Any, eps: float = _EPS) -> float:
    """Mean binary NLL / log loss."""
    yt, yp = _validate(y_true, y_prob)
    yp = np.clip(yp, eps, 1.0 - eps)
    return float(-np.mean(yt * np.log(yp) + (1.0 - yt) * np.log(1.0 - yp)))


def confidence_gap(y_true: Any, y_prob: Any) -> float:
    """Signed top-class confidence minus accuracy.

    Positive => overconfident; negative => underconfident.
    """
    yt, yp = _validate(y_true, y_prob)
    confidence = np.where(yt == 1, yp, 1.0 - yp)
    accuracy = (np.round(yp) == yt).mean()
    return float(confidence.mean() - accuracy)


def reliability_diagram(
    y_true: Any, y_prob: Any, n_bins: int = 15, strategy: str = "equal_mass"
) -> dict[str, np.ndarray]:
    """Per-bin (confidence, accuracy, count) for reliability plots."""
    yt, yp = _validate(y_true, y_prob)
    confs, accs, counts = [], [], []
    for idx in bin_indices(yp, n_bins=n_bins, strategy=strategy):
        if idx.size == 0:
            continue
        confs.append(float(yp[idx].mean()))
        accs.append(float(yt[idx].mean()))
        counts.append(int(idx.size))
    return {
        "confidence": np.asarray(confs, dtype=np.float64),
        "accuracy": np.asarray(accs, dtype=np.float64),
        "count": np.asarray(counts, dtype=np.int64),
    }


def compute_all_calibration_metrics(
    y_true: Any, y_prob: Any, n_bins: int = 15, strategy: str = "equal_width"
) -> dict[str, float]:
    """All calibration metrics in one call (used by evaluation phases).

    Defaults to ``equal_width`` so the reported ECE/MCE use the same bins as the
    Brier decomposition and stay internally consistent.
    """
    decomp = brier_decomposition(y_true, y_prob, n_bins=n_bins, strategy=strategy)
    return {
        "ece": expected_calibration_error(y_true, y_prob, n_bins=n_bins, strategy=strategy),
        "mce": maximum_calibration_error(y_true, y_prob, n_bins=n_bins, strategy=strategy),
        "brier": decomp["brier"],
        "brier_reliability": decomp["reliability"],
        "brier_resolution": decomp["resolution"],
        "brier_uncertainty": decomp["uncertainty"],
        "brier_within_bin_variance": decomp["within_bin_variance"],
        "brier_within_bin_covariance": decomp["within_bin_covariance"],
        "brier_murphy_3term": decomp["brier_murphy_3term"],
        "nll": negative_log_likelihood(y_true, y_prob),
        "confidence_gap": confidence_gap(y_true, y_prob),
    }
```

```python
The Phase 3 pilot only needs wall-clock/VRAM/power bookkeeping; Phase 10 reuses
the same helpers for the full efficiency sweep.

import platform
import statistics

def read_gpu_power_w() -> float | None:
    """Current GPU board power in watts, or None if unavailable.

    NVML ``nvmlDeviceGetPowerUsage`` is preferred; this consumer GPU/driver
    combination returns ``NotSupported`` (see Phase 0 note 8), so we fall back
    to ``nvidia-smi --query-gpu=power.draw``.
    """
    try:
        import pynvml

        pynvml.nvmlInit()
        try:
            handle = pynvml.nvmlDeviceGetHandleByIndex(0)
            mw = int(pynvml.nvmlDeviceGetPowerUsage(handle))
            if mw > 0:
                return mw / 1000.0
        finally:
            pynvml.nvmlShutdown()
    except Exception:
        pass

    try:
        res = subprocess.run(
            ["nvidia-smi", "--query-gpu=power.draw", "--format=csv,noheader,nounits"],
            capture_output=True,
            text=True,
            check=False,
            timeout=20,
        )
        token = (res.stdout or "").strip().splitlines()
        if token:
            val = float(token[0].strip())
            return val if val > 0 else None
    except Exception:
        pass
    return None

class Stopwatch:
    """Wall-clock + power/energy accounting for a named code region."""

    def __init__(self, name: str, sample_interval_s: float = 0.0) -> None:
        self.name = name
        self.sample_interval_s = sample_interval_s
        self.wall_s = 0.0
        self.energy_wh: float | None = None
        self.mean_power_w: float | None = None
        self.peak_power_w: float | None = None
        self.peak_vram_gb: float | None = None
        self.gpu_temp_c: float | None = None
        self._powers: list[float] = []

    def __enter__(self) -> "Stopwatch":
        reset_peak_vram()
        self._powers = []
        self._start_wall = time.perf_counter()
        p = read_gpu_power_w()
        if p is not None:
            self._powers.append(p)
        return self

    def sample(self) -> None:
        p = read_gpu_power_w()
        if p is not None:
            self._powers.append(p)

    def __exit__(self, *exc: object) -> None:
        self.wall_s = time.perf_counter() - self._start_wall
        self.peak_vram_gb = peak_vram_gb()
        p = read_gpu_power_w()
        if p is not None:
            self._powers.append(p)
        self.gpu_temp_c = gpu_temperature_c()
        if self._powers:
            self.mean_power_w = float(statistics.fmean(self._powers))
            self.peak_power_w = float(max(self._powers))
            self.energy_wh = self.mean_power_w * (self.wall_s / 3600.0)

    def as_dict(self) -> dict[str, Any]:
        return {
            "name": self.name,
            "wall_s": round(self.wall_s, 4),
            "wall_min": round(self.wall_s / 60.0, 4),
            "peak_vram_gb": None if self.peak_vram_gb is None else round(self.peak_vram_gb, 3),
            "mean_power_w": None if self.mean_power_w is None else round(self.mean_power_w, 2),
            "peak_power_w": None if self.peak_power_w is None else round(self.peak_power_w, 2),
            "energy_wh": None if self.energy_wh is None else round(self.energy_wh, 6),
            "gpu_temp_c": self.gpu_temp_c,
        }


@contextmanager
def timed(name: str, record: dict[str, Any] | None = None) -> Iterator[Stopwatch]:
    """Context manager yielding a :class:`Stopwatch`; merges result into ``record``."""
    sw = Stopwatch(name)
    with sw:
        yield sw
    if record is not None:
        record[name] = sw.as_dict()

def benchmark_callable(
    fn,
    *,
    n_warmup: int = 3,
    n_runs: int = 20,
) -> dict[str, float]:
    """Latency distribution (ms) for ``fn()`` with warmup."""
    for _ in range(max(0, n_warmup)):
        fn()
    latencies: list[float] = []
    for _ in range(max(1, n_runs)):
        t0 = time.perf_counter()
        fn()
        latencies.append((time.perf_counter() - t0) * 1000.0)
    arr = sorted(latencies)
    return {
        "n_runs": int(n_runs),
        "mean_ms": float(statistics.fmean(arr)),
        "median_ms": float(statistics.median(arr)),
        "p95_ms": float(arr[min(len(arr) - 1, int(0.95 * len(arr)))]),
        "min_ms": float(arr[0]),
        "max_ms": float(arr[-1]),
        "std_ms": float(statistics.pstdev(arr)) if len(arr) > 1 else 0.0,
    }
```

### `src/models/finetune.py`

```python
"""LoRA fine-tuning for instruction-format binary hate-speech classification.

The task is cast as single-token verbalizer completion: the prompt ends with
``"Answer:"`` and the model must emit ``" A"`` (hate) or ``" B"`` (non-hate).
Training masks every prompt token and supervises the single label token, so the
next-token distribution at the answer position is a calibrated two-way
classifier (see :mod:`src.models.logit_extraction`).
"""

from __future__ import annotations

import json
import logging
import math
import os
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any, Iterable

import numpy as np
import pandas as pd
import torch
from torch.utils.data import Dataset

LOG = logging.getLogger("slm-calib.finetune")

DEFAULT_PROMPT_TEMPLATE = (
    "Classify the following text as hate speech or not.\n"
    "Text: {text}\n"
    "Answer:"
)

# The answer token must be a *single* token for the two-way softmax read-out.
# " A" / " B" are single Qwen-family BPE tokens (verified by assert_single_token).
DEFAULT_VERBALIZER: dict[str, str] = {"hate": " A", "nonhate": " B"}


@dataclass
class SFTExample:
    example_id: str
    prompt: str
    label: int
    domain: str = ""
    language: str = "bn"


def build_prompt(text: str, template: str = DEFAULT_PROMPT_TEMPLATE) -> str:
    return template.format(text=" ".join(str(text).split()))


def frame_to_examples(df: pd.DataFrame, template: str = DEFAULT_PROMPT_TEMPLATE) -> list[SFTExample]:
    """Convert a processed dataframe with ``text``/``label``/``example_id``."""
    out: list[SFTExample] = []
    for row in df.itertuples(index=False):
        out.append(
            SFTExample(
                example_id=str(getattr(row, "example_id")),
                prompt=build_prompt(row.text, template),
                label=int(row.label),
                domain=str(getattr(row, "domain", "")),
                language=str(getattr(row, "language", "bn")),
            )
        )
    return out


class VerbalizedSFTDataset(Dataset):
    """Prompt + single label token, with prompt tokens masked out of the loss."""

    def __init__(
        self,
        examples: Iterable[SFTExample],
        tokenizer: Any,
        max_len: int = 256,
        template: str = DEFAULT_PROMPT_TEMPLATE,
    ) -> None:
        self.examples = list(examples)
        self.tokenizer = tokenizer
        self.max_len = max_len
        self.template = template
        self.label_token_ids = label_token_ids(tokenizer, DEFAULT_VERBALIZER)

    def __len__(self) -> int:
        return len(self.examples)

    def __getitem__(self, idx: int) -> dict[str, Any]:
        ex = self.examples[idx]
        prompt_ids = self.tokenizer(ex.prompt, add_special_tokens=False)["input_ids"]
        answer_ids = [self.label_token_ids[int(ex.label)]]
        # Truncate the prompt from the LEFT so the answer slot is always kept.
        budget = self.max_len - len(answer_ids)
        if len(prompt_ids) > budget:
            prompt_ids = prompt_ids[:1] + prompt_ids[-(budget - 1) :]
        input_ids = prompt_ids + answer_ids
        labels = [-100] * len(prompt_ids) + answer_ids
        return {
            "input_ids": input_ids,
            "labels": labels,
            "attention_mask": [1] * len(input_ids),
            "example_id": ex.example_id,
            "label": int(ex.label),
        }


class SFTCollator:
    """Right-pad a batch and pad labels with the model's ignore index."""

    def __init__(self, pad_token_id: int, label_pad_token_id: int = -100) -> None:
        self.pad_token_id = pad_token_id
        self.label_pad_token_id = label_pad_token_id

    def __call__(self, features: list[dict[str, Any]]) -> dict[str, Any]:
        max_len = max(len(f["input_ids"]) for f in features)
        batch = {
            "input_ids": [],
            "attention_mask": [],
            "labels": [],
            "example_id": [f["example_id"] for f in features],
            "label": [f["label"] for f in features],
        }
        for f in features:
            pad = max_len - len(f["input_ids"])
            batch["input_ids"].append(f["input_ids"] + [self.pad_token_id] * pad)
            batch["attention_mask"].append(f["attention_mask"] + [0] * pad)
            batch["labels"].append(f["labels"] + [self.label_pad_token_id] * pad)
        out = {
            k: (torch.tensor(v, dtype=torch.long) if k != "example_id" else v)
            for k, v in batch.items()
        }
        out["labels"][out["attention_mask"] == 0] = self.label_pad_token_id
        return out


# ---------------------------------------------------------------------------
# Verbalizer helpers
# ---------------------------------------------------------------------------

def encode_label_token(tokenizer: Any, token: str) -> int:
    """Return the id of the *first* token of ``token``, asserting it is single."""
    ids = tokenizer.encode(token, add_special_tokens=False)
    if len(ids) != 1:
        raise ValueError(
            f"Verbalizer token {token!r} maps to {len(ids)} tokens {ids}; "
            "the two-way read-out requires single-token labels."
        )
    return int(ids[0])


def label_token_ids(tokenizer: Any, verbalizer: dict[str, str] | None = None) -> dict[int, int]:
    """Map class index (0 = non-hate, 1 = hate) to token id."""
    v = verbalizer or DEFAULT_VERBALIZER
    return {0: encode_label_token(tokenizer, v["nonhate"]), 1: encode_label_token(tokenizer, v["hate"])}


def assert_single_token_verbalizer(tokenizer: Any, verbalizer: dict[str, str] | None = None) -> dict[str, int]:
    """T-style guard: verbalizer tokens must be distinct and single-token."""
    v = verbalizer or DEFAULT_VERBALIZER
    hate = encode_label_token(tokenizer, v["hate"])
    nonhate = encode_label_token(tokenizer, v["nonhate"])
    if hate == nonhate:
        raise ValueError(f"Verbalizer tokens collide: {v['hate']!r} and {v['nonhate']!r} -> {hate}")
    return {"hate": hate, "nonhate": nonhate}

# Training
# ---------------------------------------------------------------------------

@dataclass
class FinetuneResult:
    adapter_path: str
    merged_path: str
    history: list[dict[str, float]] = field(default_factory=list)
    best_eval_loss: float | None = None
    best_checkpoint: str | None = None
    n_train: int = 0
    n_id_val: int = 0
    loss_decreased: bool = False
    final_train_loss: float | None = None
    peak_vram_gb: float | None = None
    wall_s: float | None = None
    trainable_params: int = 0
    total_params: int = 0
    label_token_ids: dict[str, int] = field(default_factory=dict)
    max_seq_len: int = 0


def load_base_model(hf_id: str, dtype: str = "bfloat16", device_map: str | None = "auto"):
    """Load a causal LM in the requested dtype."""
    from transformers import AutoModelForCausalLM

    torch_dtype = {
        "bfloat16": torch.bfloat16,
        "bf16": torch.bfloat16,
        "float16": torch.float16,
        "fp16": torch.float16,
        "float32": torch.float32,
        "fp32": torch.float32,
    }[dtype.lower()]

    kwargs: dict[str, Any] = {"torch_dtype": torch_dtype}
    if device_map:
        kwargs["device_map"] = device_map
    model = AutoModelForCausalLM.from_pretrained(hf_id, **kwargs)
    model.config.use_cache = False
    return model


def load_tokenizer(hf_id: str):
    from transformers import AutoTokenizer

    tok = AutoTokenizer.from_pretrained(hf_id)
    if tok.pad_token is None:
        tok.pad_token = tok.eos_token
    tok.padding_side = "right"
    return tok


def finetune_lora(config: dict[str, Any], train_df: pd.DataFrame, val_df: pd.DataFrame) -> FinetuneResult:
    """LoRA fine-tune a causal LM with completion-only loss.

    ``config`` keys (mirrors ``configs/pilot/default.yaml``)::

        model.hf_id, model.dtype, model.verbalizer.{hate,nonhate}
        train.{rank,alpha,dropout,target_modules,lr,epochs,batch_size,grad_accum,
               warmup_ratio,weight_decay,grad_checkpointing,max_grad_norm}
        data.max_len, data.prompt_template
        experiment.{name,seed,output_dir}

    Returns a :class:`FinetuneResult`; the merged model is written to
    ``<output_dir>/merged``.
    """
    from peft import LoraConfig, TaskType, get_peft_model
    from src.metrics.efficiency import Stopwatch
    from transformers import Trainer, TrainingArguments, set_seed as hf_set_seed

    model_cfg = config["model"]
    train_cfg = config["train"]
    data_cfg = config.get("data", {})
    exp_cfg = config["experiment"]

    seed = int(exp_cfg["seed"])
    hf_set_seed(seed)

    output_dir = Path(exp_cfg["output_dir"])
    output_dir.mkdir(parents=True, exist_ok=True)

    max_len = int(data_cfg.get("max_len", 256))
    template = str(data_cfg.get("prompt_template", DEFAULT_PROMPT_TEMPLATE))
    verbalizer = dict(model_cfg.get("verbalizer", DEFAULT_VERBALIZER))

    sw = Stopwatch("finetune")
    with sw:
        tokenizer = load_tokenizer(model_cfg["hf_id"])
        token_ids = assert_single_token_verbalizer(tokenizer, verbalizer)
        LOG.info("Verbalizer token ids: %s", token_ids)

        model = load_base_model(model_cfg["hf_id"], str(model_cfg.get("dtype", "bfloat16")))
        model.config.pad_token_id = tokenizer.pad_token_id

        lora_config = LoraConfig(
            r=int(train_cfg.get("rank", 16)),
            lora_alpha=int(train_cfg.get("alpha", 32)),
            lora_dropout=float(train_cfg.get("dropout", 0.05)),
            target_modules=list(train_cfg["target_modules"]),
            bias="none",
            task_type=TaskType.CAUSAL_LM,
        )
        model = get_peft_model(model, lora_config)
        trainable, total = model.get_nb_trainable_parameters()
        model.print_trainable_parameters()
        if bool(train_cfg.get("grad_checkpointing", True)):
            # Without this the checkpointed blocks receive inputs that do not require
            # grad, no graph is built, and backward() raises "element 0 of tensors does
            # not require grad and does not have a grad_fn".
            model.enable_input_require_grads()

        train_ds = VerbalizedSFTDataset(
            frame_to_examples(train_df, template), tokenizer, max_len=max_len, template=template
        )
        val_ds = VerbalizedSFTDataset(
            frame_to_examples(val_df, template), tokenizer, max_len=max_len, template=template
        )
        collator = SFTCollator(pad_token_id=tokenizer.pad_token_id)

        epochs = float(train_cfg.get("epochs", 1))
        batch_size = int(train_cfg.get("batch_size", 2))
        grad_accum = int(train_cfg.get("grad_accum", 8))
        # A short run must still emit loss points: with logging_steps larger than the
        # number of optimiser updates, log_history contains no "loss" entry at all.
        steps_per_epoch = max(1, math.ceil(len(train_ds) / (batch_size * grad_accum)))
        total_updates = max(1, int(steps_per_epoch * epochs))
        configured_logging = int(train_cfg.get("logging_steps", 10))
        logging_steps = max(1, min(configured_logging, total_updates // 5 or 1))
        args = TrainingArguments(
            output_dir=str(output_dir / "checkpoints"),
            num_train_epochs=epochs,
            per_device_train_batch_size=batch_size,
            per_device_eval_batch_size=int(train_cfg.get("eval_batch_size", 4)),
            gradient_accumulation_steps=grad_accum,
            gradient_checkpointing=bool(train_cfg.get("grad_checkpointing", True)),
            learning_rate=float(train_cfg["lr"]),
            weight_decay=float(train_cfg.get("weight_decay", 0.0)),
            warmup_ratio=float(train_cfg.get("warmup_ratio", 0.03)),
            max_grad_norm=float(train_cfg.get("max_grad_norm", 1.0)),
            lr_scheduler_type=str(train_cfg.get("lr_scheduler", "cosine")),
            logging_steps=logging_steps,
            save_strategy=str(train_cfg.get("save_strategy", "epoch")),
            save_total_limit=int(train_cfg.get("save_total_limit", 2)),
            eval_strategy=str(train_cfg.get("eval_strategy", "epoch")),
            load_best_model_at_end=True,
            metric_for_best_model=str(train_cfg.get("metric_for_best_model", "eval_loss")),
            greater_is_better=False,
            bf16=bool(train_cfg.get("bf16", True)),
            seed=seed,
            data_seed=seed,
            report_to=[],
            remove_unused_columns=False,
            dataloader_num_workers=int(train_cfg.get("dataloader_workers", 0)),
        )

        trainer = Trainer(
            model=model,
            args=args,
            train_dataset=train_ds,
            eval_dataset=val_ds,
            data_collator=collator,
        )
        train_out = trainer.train()
        metrics = trainer.evaluate()
        trainer.save_model(str(output_dir / "adapter"))
        tokenizer.save_pretrained(str(output_dir / "adapter"))

        history = [
            {k: (float(v) if isinstance(v, (int, float)) else v) for k, v in step.items()}
            for step in trainer.state.log_history
        ]
        best_ckpt = getattr(trainer.state, "best_model_checkpoint", None)

        # Merge adapter into the base weights -> standalone model for PTQ.
        merged_dir = output_dir / "merged"
        merged = trainer.model.merge_and_unload()
        merged.config.use_cache = True
        merged.save_pretrained(str(merged_dir), safe_serialization=True)
        tokenizer.save_pretrained(str(merged_dir))

    train_losses = [h["loss"] for h in history if "loss" in h]
    loss_decreased = bool(
        len(train_losses) >= 2 and train_losses[-1] < train_losses[0] and not any(
            not math.isfinite(v) for v in train_losses
        )
    )
    (output_dir / "train_metrics.json").write_text(
        json.dumps(
            {
                "train_runtime_s": float(getattr(train_out.metrics, "train_runtime", sw.wall_s)),
                "train_samples_per_second": float(
                    getattr(train_out.metrics, "train_samples_per_second", 0.0) or 0.0
                ),
                "eval": {k: float(v) for k, v in metrics.items() if isinstance(v, (int, float))},
                "log_history": history,
                "label_token_ids": token_ids,
                "best_checkpoint": best_ckpt,
                "total_updates": total_updates,
                "logging_steps": logging_steps,
                "peak_vram_gib": sw.peak_vram_gb,
                "wall_s": sw.wall_s,
                "energy_wh": sw.energy_wh,
                "trainable_params": trainable,
                "total_params": total,
                "n_train": len(train_ds),
                "n_id_val": len(val_ds),
                "max_seq_len": max_len,
            },
            indent=2,
        ),
        encoding="utf-8",
    )

    return FinetuneResult(
        adapter_path=str(output_dir / "adapter"),
        merged_path=str(merged_dir),
        history=history,
        best_eval_loss=float(metrics.get("eval_loss")) if metrics.get("eval_loss") is not None else None,
        n_train=len(train_ds),
        n_id_val=len(val_ds),
        loss_decreased=loss_decreased,
        best_checkpoint=str(best_ckpt) if best_ckpt else None,
        final_train_loss=train_losses[-1] if train_losses else None,
        peak_vram_gb=sw.peak_vram_gb,
        wall_s=sw.wall_s,
        trainable_params=int(trainable),
        total_params=int(total),
        label_token_ids=token_ids,
        max_seq_len=max_len,
    )


def default_target_modules(hf_id: str) -> list[str]:

def subsample(df: pd.DataFrame, n: int | None, seed: int = 42) -> pd.DataFrame:
    """Deterministic stratified subsample (stratified on ``label``) used to cap pilot cost.

    Allocation uses the largest-remainder method so the class ratio is preserved
    and the row count sums *exactly* to ``n``.  Written without ``groupby.apply``
    because recent pandas releases drop the grouping column from the result.
    """
    if not n or n >= len(df):
        return df.reset_index(drop=True)
    n = int(n)
    if "label" not in df.columns:
        raise ValueError("subsample() requires a 'label' column for stratification")
    if n < int(df["label"].nunique()):
        # Too few rows to keep every class; stratification is impossible.
        return df.sample(n=n, random_state=seed).reset_index(drop=True)

    counts = df["label"].value_counts().sort_index()
    exact = counts.to_numpy(dtype=np.float64) * (n / len(df))
    frac = exact - np.floor(exact)
    shares = np.clip(np.floor(exact).astype(np.int64), 1, counts.to_numpy())

    order = np.argsort(-frac, kind="stable")
    guard = 0
    while shares.sum() < n and guard < 10 * n:
        for i in order:
            if shares[i] < counts.to_numpy()[i]:
                shares[i] += 1
                break
        guard += 1
    guard = 0
    while shares.sum() > n and guard < 10 * n:
        for i in order[::-1]:
            if shares[i] > 1:
                shares[i] -= 1
                break
        guard += 1

    picked = pd.concat(
        [
            df[df["label"] == label].sample(int(k), random_state=seed)
            for label, k in zip(counts.index, shares)
            if k > 0
        ],
        ignore_index=True,
    )
    return picked.sample(frac=1.0, random_state=seed).reset_index(drop=True)
```

### `src/metrics/aggregate.py`

```python
"""One-call aggregation of every metric from a canonical logits frame.

Phase 3 uses this to prove T3.9 ("all metrics computable on pilot logits");
Phase 8 reuses it for the full evaluation sweep, so the metric definition is
written exactly once.
"""

from __future__ import annotations

from typing import Any

import numpy as np
import pandas as pd

from src.metrics.calibration import compute_all_calibration_metrics
from src.metrics.selective import compute_all_selective_metrics
from src.metrics.task_metrics import agreement_rate, compute_all_task_metrics


def _probability_column(df: pd.DataFrame) -> np.ndarray:
    """P(hate) from whichever probability column the frame carries."""
    for col in ("prob_hate", "y_prob", "prob_positive", "probability"):
        if col in df.columns:
            return df[col].to_numpy(dtype=np.float64)
    raise ValueError(f"No positive-class probability column in frame: {list(df.columns)}")


def evaluate_logits_frame(
    df: pd.DataFrame,
    *,
    ece_bins: int = 15,
    ece_strategy: str = "equal_width",
    aurc_method: str = "riemann",
) -> dict[str, Any]:
    """Compute task + calibration + selective metrics from a logits frame.

    ``prediction`` is used when present (so quantization variants are scored on
    their own decisions); otherwise it is thresholded at ``p = 0.5``.
    """
    y_true = df["y_true"].to_numpy(dtype=np.int64)
    y_prob = _probability_column(df)
    if "prediction" in df.columns:
        y_pred = df["prediction"].to_numpy(dtype=np.int64)
    else:
        y_pred = (y_prob >= 0.5).astype(np.int64)

    out: dict[str, Any] = {}
    out.update(compute_all_task_metrics(y_true, y_pred, y_prob))
    out.update(compute_all_calibration_metrics(y_true, y_prob, n_bins=ece_bins, strategy=ece_strategy))
    sel = compute_all_selective_metrics(y_true, y_prob)
    if aurc_method != "riemann":
        from src.metrics.selective import aurc as _aurc
        from src.metrics.selective import e_aurc as _e_aurc

        sel["aurc"] = _aurc(y_true, y_prob, aurc_method)
        sel["e_aurc"] = _e_aurc(y_true, y_prob, aurc_method)
    out.update(sel)
    # bd_shs derives `domain` from the hate subtype, so every LOCO target domain is
    # single-class (slander/callToViolence/gender/religion are all hate; non_hate is
    # all non-hate).  Rank metrics such as AUROC are undefined there; flag it rather
    # than letting -inf leak into the results table.
    out["single_class_eval_set"] = bool(np.unique(y_true).size < 2)
    if "split" in df.columns:
        out["splits"] = sorted(set(df["split"].astype(str)))
    return out


def compare_precision_frames(
    reference: pd.DataFrame, variant: pd.DataFrame
) -> dict[str, Any]:
    """Accuracy / agreement deltas of a quantized variant against a reference."""
    from src.metrics.task_metrics import bootstrap_accuracy_delta

    merged = reference.merge(variant, on="example_id", suffixes=("_ref", "_var"), how="inner")
    if len(merged) == 0:
        raise ValueError("No overlapping example_id between reference and variant logits")
    if "y_true_ref" in merged and "y_true_var" in merged:
        mismatch = int((merged["y_true_ref"] != merged["y_true_var"]).sum())
    else:
        mismatch = 0

    y_true = merged["y_true_ref"].to_numpy(dtype=np.int64)
    out = {
        "n_aligned": int(len(merged)),
        "label_mismatch": mismatch,
        "acc_reference": float((merged["prediction_ref"] == y_true).mean()),
        "acc_variant": float((merged["prediction_var"] == y_true).mean()),
        "agreement_rate": agreement_rate(merged["prediction_ref"], merged["prediction_var"]),
        "mean_abs_prob_hate_delta": float(
            np.mean(np.abs(merged["prob_hate_ref"] - merged["prob_hate_var"]))
        ),
        "max_abs_prob_hate_delta": float(
            np.max(np.abs(merged["prob_hate_ref"] - merged["prob_hate_var"]))
        ),
    }
    out["accuracy_drop"] = out["acc_reference"] - out["acc_variant"]
    out["bootstrap"] = bootstrap_accuracy_delta(
        y_true, merged["prediction_ref"], merged["prediction_var"], n_boot=1000, seed=42
    )
    return out


def metrics_to_record(metrics: dict[str, Any], **context: Any) -> dict[str, Any]:
    """Flatten a metrics dict (plus context) into a JSON-safe record."""
    record: dict[str, Any] = {**context}
    for key, value in metrics.items():
        if isinstance(value, (str, bool, int)):
            record[key] = value
        elif isinstance(value, (list, tuple)):
            record[key] = list(value)
        elif isinstance(value, float) and not np.isfinite(value):
            record[key] = None
        else:
            record[key] = value
    return record


def metrics_frame(records: list[dict[str, Any]]) -> pd.DataFrame:
    """Assemble per-(model, precision, split) metric rows into one table."""
    flat = [metrics_to_record(r) for r in records]
    if not flat:
        return pd.DataFrame()
    frame = pd.DataFrame(flat)
    leading = [c for c in ("model", "precision", "split", "language", "loco", "seed") if c in frame.columns]
    rest = [c for c in frame.columns if c not in leading]
    return frame[leading + sorted(rest)]
```

```python
def load_hatexplain(raw_dir: str | Path = "data/raw/hatexplain") -> pd.DataFrame:
    """Load HateXplain (20,148 posts) from the official JSON dump.

    Majority-vote label -> binary hate (hatespeech=1; normal/offensive=0).
    Domain axis: majority targeted community from annotators.
    """
    raw_dir = Path(raw_dir)
    path = raw_dir / "hatexplain.json"
    if not path.exists():
        raise FileNotFoundError(f"HateXplain JSON not found: {path}")
    obj = json.loads(path.read_text(encoding="utf-8"))
    if not isinstance(obj, dict):
        raise ValueError("HateXplain JSON must be a dict keyed by post_id")

    rows: list[dict[str, Any]] = []
    for pid, item in obj.items():
        annotators = item.get("annotators") or []
        if not annotators:
            continue
        label_votes = [a.get("label") for a in annotators if isinstance(a, dict)]
        label_votes = [v for v in label_votes if v]
        if not label_votes:
            continue
        majority = Counter(label_votes).most_common(1)[0][0]
        label = 1 if majority == "hatespeech" else 0

        target_votes: list[str] = []
        for a in annotators:
            if not isinstance(a, dict):
                continue
            t = a.get("target")
            if t is None:
                continue
            if isinstance(t, list):
                target_votes.extend([str(x) for x in t if x and str(x).lower() != "none"])
            else:
                tv = str(t)
                if tv.lower() != "none":
                    target_votes.append(tv)
        domain = Counter(target_votes).most_common(1)[0][0] if target_votes else "unspecified"

        tokens = item.get("post_tokens") or []
        text = " ".join(tokens) if tokens else ""
        if not text.strip():
            continue
        platform = "twitter" if str(pid).endswith("_twitter") else (
            "gab" if str(pid).endswith("_gab") else "unknown"
        )
        rows.append(
            {
                "example_id": f"hateexplain_{pid}",
                "text": text,
                "label": label,
                "majority_label_raw": majority,
                "domain": domain,
                "platform": platform,
                "language": "en",
                "dataset": "hatexplain",
                "source_split": "full",
            }
        )
    df = pd.DataFrame(rows)
    print(f"HateXplain loaded: {len(df)} samples")
    print(f"  label prior: {df['label'].mean():.4f}")
    print(f"  domains ({df['domain'].nunique()}): {df['domain'].value_counts().head(12).to_dict()}")
    return df

def _standardize_toxigen(df: pd.DataFrame) -> pd.DataFrame:
    cols = {c.lower(): c for c in df.columns}
    # HF train split uses `generation` (generated claim) as the text field.
    text_col = next(
        (cols[k] for k in ("text", "sentence", "content", "generation") if k in cols),
        None,
    )
    if text_col is None:
        raise ValueError(f"ToxiGen: no text column; cols={list(df.columns)}")

    label = None
    # Prefer human annotations when present; else prompt_label / toxicity scores.
    if "toxicity_human" in cols:
        toxic = pd.to_numeric(df[cols["toxicity_human"]], errors="coerce").fillna(0)
        label = (toxic > 0.5).astype(int)
    elif "prompt_label" in cols:
        raw = df[cols["prompt_label"]]
        if raw.dtype == object:
            mapping = {
                "hate": 1, "toxic": 1, "toxigen": 1, "1": 1, 1: 1,
                "benign": 0, "non-toxic": 0, "normal": 0, "0": 0, 0: 0,
            }
            label = raw.map(lambda v: mapping.get(str(v).lower(), 0)).astype(int)
        else:
            label = (pd.to_numeric(raw, errors="coerce").fillna(0) > 0).astype(int)
    elif "toxicity_ai" in cols:
        toxic = pd.to_numeric(df[cols["toxicity_ai"]], errors="coerce").fillna(0)
        label = (toxic > 0.5).astype(int)
    elif "toxicity" in cols:
        toxic = pd.to_numeric(df[cols["toxicity"]], errors="coerce").fillna(0)
        label = (toxic > 0.5).astype(int)
    elif "label" in cols:
        raw = df[cols["label"]]
        if raw.dtype == object:
            mapping = {
                "hate": 1, "toxic": 1, "toxigen": 1, "1": 1, 1: 1,
                "benign": 0, "non-toxic": 0, "normal": 0, "0": 0, 0: 0,
            }
            label = raw.map(lambda v: mapping.get(str(v).lower(), 0)).astype(int)
        else:
            label = (pd.to_numeric(raw, errors="coerce").fillna(0) > 0).astype(int)
    else:
        raise ValueError(f"ToxiGen: no label/toxicity column; cols={list(df.columns)}")

    domain_col = None
    for cand in ("group", "predicted_group", "target_group", "identity"):
        if cand in cols:
            domain_col = cols[cand]
            break
    domain = (
        df[domain_col].astype(str).fillna("unspecified")
        if domain_col
        else pd.Series(["unspecified"] * len(df))
    )
    domain = domain.replace({"nan": "unspecified", "None": "unspecified", "": "unspecified"})

    out = pd.DataFrame(
        {
            "example_id": [f"toxigen_{i}" for i in range(len(df))],
            "text": df[text_col].astype(str),
            "label": label.astype(int),
            "domain": domain.values,
            "language": "en",
            "dataset": "toxigen",
            "source_split": "full",
        }
    )
    print(f"ToxiGen loaded: {len(out)} samples; prior={out['label'].mean():.4f}")
    print(f"  domains ({out['domain'].nunique()}): {out['domain'].value_counts().head(10).to_dict()}")
    return out


def subsample_to_match(
    df: pd.DataFrame,
    *,
    target_n: int,
    target_prior: float,
    seed: int = 42,
    text_col: str = "text",
    label_col: str = "label",
) -> pd.DataFrame:
    """Stratified subsample so size is close to target_n and prior to target_prior.

    Within each label stratum we keep as many rows as the class-prior target
    allows, then top-up/down to hit target_n as closely as possible.
    """
    if len(df) == 0:
        raise ValueError("subsample_to_match: empty dataframe")
    rng = np.random.default_rng(seed)
    n0 = max(1, min(len(df), target_n))
    n_pos = int(round(n0 * target_prior))
    n_pos = max(0, min(n0, n_pos))
    n_neg = n0 - n_pos

    pos = df[df[label_col] == 1]
    neg = df[df[label_col] == 0]
    if len(pos) < n_pos or len(neg) < n_neg:
        # Fall back to proportional sampling of what we have.
        total = len(df)
        n_pos = min(n_pos, len(pos))
        n_neg = min(n_neg, len(neg))
        # If still short overall, scale up to available size
        if n_pos + n_neg < min(target_n, total):
            # use as much as possible while keeping prior as close as we can
            avail = len(pos) + len(neg)
            n_pos = min(len(pos), int(round(avail * target_prior)))
            n_neg = min(len(neg), avail - n_pos)

    def _take(frame: pd.DataFrame, k: int) -> pd.DataFrame:
        if k <= 0 or len(frame) == 0:
            return frame.iloc[0:0]
        if k >= len(frame):
            return frame
        idx = rng.choice(len(frame), size=k, replace=False)
        return frame.iloc[np.sort(idx)]

    sampled = pd.concat([_take(pos, n_pos), _take(neg, n_neg)], ignore_index=True)
    sampled = sampled.sample(frac=1.0, random_state=seed).reset_index(drop=True)
    print(
        f"Subsampled English control: {len(sampled)} rows "
        f"(prior={sampled[label_col].mean():.4f}, target_n={target_n}, target_prior={target_prior})"
    )
    return sampled


def load_english_control(
    dataset_name: str = "auto",
    *,
    target_n: int | None = None,
    target_prior: float | None = None,
    seed: int = 42,
    prefer: str = "toxigen",
) -> tuple[pd.DataFrame, dict[str, Any]]:
    """Load the English control dataset and match BD-SHS size/class prior.

    Decision (Phase 1):
      - HateXplain has rich target-community metadata but only ~20k rows,
        so it CANNOT satisfy T1.10 (size within 10% of BD-SHS ~50k).
      - ToxiGen is large enough and has identity-group metadata for LOCO.
      - Therefore the primary English control is ToxiGen (subsampled).
      - HateXplain is retained as a secondary English metadata-rich corpus.
    """
    bd_prior = 24156 / 50281
    target_n = target_n if target_n is not None else BD_SHS_EXPECTED_N
    target_prior = target_prior if target_prior is not None else bd_prior

    decision: dict[str, Any] = {
        "chosen": None,
        "candidates": {},
        "target_n": target_n,
        "target_prior": target_prior,
        "justification": "",
    }

    hatexplain_df: pd.DataFrame | None = None
    try:
        hatexplain_df = load_hatexplain()
        decision["candidates"]["hatexplain"] = {
            "n": int(len(hatexplain_df)),
            "prior": float(hatexplain_df["label"].mean()),
            "n_domains": int(hatexplain_df["domain"].nunique()),
            "role": "secondary_metadata_rich",
        }
    except Exception as e:
        decision["candidates"]["hatexplain"] = {"error": str(e)}

    if dataset_name in ("auto", "toxigen", prefer):
        try:
            tox = load_toxigen()
        except Exception as e:
            print(f"ToxiGen load failed: {e}")
            tox = None
        if tox is not None:
            decision["candidates"]["toxigen"] = {
                "n": int(len(tox)),
                "prior": float(tox["label"].mean()),
                "n_domains": int(tox["domain"].nunique()),
                "role": "primary_control",
            }
            matched = subsample_to_match(
                tox, target_n=target_n, target_prior=target_prior, seed=seed
            )
            # Size / prior acceptance
            size_ok = abs(len(matched) - target_n) / target_n <= 0.10
            prior_ok = abs(matched["label"].mean() - target_prior) <= 0.02
            decision["chosen"] = "toxigen"
            decision["matched_n"] = int(len(matched))
            decision["matched_prior"] = float(matched["label"].mean())
            decision["size_ok"] = bool(size_ok)
            decision["prior_ok"] = bool(prior_ok)
            decision["justification"] = (
                "ToxiGen selected as primary English control: large enough to "
                "subsample to BD-SHS size (T1.10), binary toxicity labels, and "
                "identity-group metadata for LOCO-style domain shift. HateXplain "
                "is kept as a secondary English corpus with richer target-community "
                "labels but is too small (~20k) to match BD-SHS size."
            )
            print("English control decision:")
            print(json.dumps(decision, indent=2))
            return matched, decision

    # Fallback: HateXplain matched as closely as possible (may fail T1.10 size).
    if hatexplain_df is not None:
        matched = subsample_to_match(
            hatexplain_df,
            target_n=min(target_n, len(hatexplain_df)),
            target_prior=target_prior,
            seed=seed,
        )
        decision["chosen"] = "hatexplain_fallback"
        decision["matched_n"] = int(len(matched))
        decision["matched_prior"] = float(matched["label"].mean())
        decision["size_ok"] = False
        decision["prior_ok"] = abs(matched["label"].mean() - target_prior) <= 0.02
        decision["justification"] = (
            "Fallback to HateXplain; size will not meet T1.10. Prefer ToxiGen."
        )
        return matched, decision

    raise RuntimeError("No English control dataset could be loaded")


def write_english_control_decision(
    decision: dict[str, Any], path: str | Path
) -> Path:
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    def _default(o: Any) -> Any:
        if isinstance(o, (np.integer,)):
            return int(o)
        if isinstance(o, (np.floating,)):
            return float(o)
        if isinstance(o, (np.bool_,)):
            return bool(o)
        if isinstance(o, np.ndarray):
            return o.tolist()
        return str(o)

    path.write_text(json.dumps(decision, indent=2, default=_default), encoding="utf-8")
    return path
```

```python
from __future__ import annotations

from dataclasses import dataclass, field

@dataclass
class CheckResult:
    name: str
    passed: bool
    detail: str = ""
    extras: dict[str, Any] = field(default_factory=dict)

    def as_dict(self) -> dict[str, Any]:
        return {

            "passed": self.passed,
            "detail": self.detail,
            **self.extras,
        }


def check_gpu_min_vram(min_vram_gb: float = 7.5) -> CheckResult:
    """Verify CUDA GPU is present with at least min_vram_gb VRAM."""
    try:
        import torch
    except ImportError:
        return CheckResult("gpu_vram", False, "torch not installed")
    if not torch.cuda.is_available():
        return CheckResult("gpu_vram", False, "CUDA not available")
    props = torch.cuda.get_device_properties(0)
    total_bytes = getattr(props, "total_memory", None) or props.total_mem
    vram_gb = round(total_bytes / 1e9, 2)
    name = torch.cuda.get_device_name(0)
    passed = vram_gb >= min_vram_gb
    return CheckResult(
        "gpu_vram",
        passed,
        f"{name}: {vram_gb} GB (need >= {min_vram_gb} GB)",
        {"gpu_name": name, "vram_gb": vram_gb},
    )


def check_required_imports(modules: list[str]) -> CheckResult:
    """Verify a list of import names are importable."""
    missing = []
    versions: dict[str, str] = {}
    import importlib

    for mod in modules:
        try:
            m = importlib.import_module(mod)
            versions[mod] = getattr(m, "__version__", "unknown")
        except ImportError:
            missing.append(mod)
    passed = not missing
    detail = "all present" if passed else f"missing: {missing}"
    return CheckResult("required_imports", passed, detail, {"versions": versions})
```

### `src/models/logit_extraction.py`

```python
"""Per-example logit / probability extraction for HF causal LMs.

Two-way read-out: the model's next-token distribution at the answer position is
restricted to the two verbalizer tokens (``" A"`` hate, ``" B"`` non-hate) and
renormalised.  This keeps the output distribution a proper binary probability
vector, which every downstream metric assumes.
"""

from __future__ import annotations

import logging
import time
from pathlib import Path
from typing import Any, Callable, Iterable

import numpy as np
import pandas as pd
import torch

from src.models.finetune import (
    DEFAULT_PROMPT_TEMPLATE,
    DEFAULT_VERBALIZER,
    build_prompt,
    encode_label_token,
)

LOG = logging.getLogger("slm-calib.logits")

REQUIRED_COLUMNS = [
    "example_id",
    "split",
    "y_true",
    "logit_hate",
    "logit_nonhate",
    "prob_hate",
    "prob_nonhate",
    "prediction",
    "confidence",
]

_LOGIT_COLUMNS = [
    "logit_hate",
    "logit_nonhate",
    "prob_hate",
    "prob_nonhate",
    "prediction",
    "confidence",
    "correct",
    "margin",
    "entropy",
]


def build_eval_examples(df: pd.DataFrame, template: str = DEFAULT_PROMPT_TEMPLATE) -> list[dict[str, Any]]:
    return [
        {
            "example_id": str(getattr(row, "example_id")),
            "prompt": build_prompt(row.text, template),
            "y_true": int(row.label),
            "domain": str(getattr(row, "domain", "")),
        }
        for row in df.itertuples(index=False)
    ]





def answer_logits_batched(
    model: Any,
    tokenizer: Any,
    prompts: list[str],
    hate_token_id: int,
    nonhate_token_id: int,
    max_len: int = 256,
) -> tuple[np.ndarray, np.ndarray]:
    """Next-token logits at the answer position, restricted to the two labels.

    The answer position is the *last real* token of each sequence.  Batches are
    right-padded, so that index is ``attention_mask.sum(1) - 1`` and differs per
    row; indexing ``logits[:, -1, :]`` reads the padded slot of every row shorter
    than the longest prompt in the batch, which silently corrupts the labels.

    Truncation is forced to the left so an over-long prompt loses its prefix
    rather than the trailing verbalizer cue that carries the answer position.

    Returns ``(logit_hate, logit_nonhate)`` as float32 arrays.
    """
    device = next(model.parameters()).device
    previous_side = getattr(tokenizer, "truncation_side", "right")
    tokenizer.truncation_side = "left"
    try:
        encoded = tokenizer(
            prompts,
            add_special_tokens=False,
            padding=True,
            truncation=True,
            max_length=max_len,
            return_tensors="pt",
        )
    finally:
        tokenizer.truncation_side = previous_side
    encoded = {k: v.to(device) for k, v in encoded.items()}
    with torch.no_grad():
        out = model(**encoded, use_cache=False)
    last_real = encoded["attention_mask"].sum(dim=1) - 1
    rows = torch.arange(out.logits.shape[0], device=out.logits.device)
    answer = out.logits[rows, last_real, :]
    return (
        answer[:, hate_token_id].float().cpu().numpy(),
        answer[:, nonhate_token_id].float().cpu().numpy(),
    )


def softmax_pair(logit_hate: np.ndarray, logit_nonhate: np.ndarray) -> np.ndarray:
    """Renormalised two-way probabilities, shape ``(N, 2)`` ordered [nonhate, hate]."""
    stacked = np.stack(
        [np.asarray(logit_nonhate, dtype=np.float64), np.asarray(logit_hate, dtype=np.float64)],
        axis=1,
    )
    shifted = stacked - stacked.max(axis=1, keepdims=True)
    exp = np.exp(shifted)
    return exp / exp.sum(axis=1, keepdims=True)


def probs_to_frame(
    example_ids: Iterable[str],
    split: str,
    y_true: Iterable[int],
    logit_hate: np.ndarray,
    logit_nonhate: np.ndarray,
    extra: dict[str, Iterable[Any]] | None = None,
) -> pd.DataFrame:
    """Assemble the canonical logits dataframe from raw two-way logits."""
    probs = softmax_pair(logit_hate, logit_nonhate)
    prob_nonhate, prob_hate = probs[:, 0], probs[:, 1]
    prediction = (prob_hate >= prob_nonhate).astype(np.int64)
    confidence = np.maximum(prob_hate, prob_nonhate)
    y = np.asarray(list(y_true), dtype=np.int64)
    eps = 1e-12
    entropy = -(prob_hate * np.log(prob_hate + eps) + prob_nonhate * np.log(prob_nonhate + eps))

    frame = pd.DataFrame(
        {
            "example_id": list(example_ids),
            "split": split,
            "y_true": y,
            "logit_hate": np.asarray(logit_hate, dtype=np.float64),
            "logit_nonhate": np.asarray(logit_nonhate, dtype=np.float64),
            "prob_hate": prob_hate,
            "prob_nonhate": prob_nonhate,
            "prediction": prediction,
            "confidence": confidence,
            "correct": (prediction == y).astype(np.int64),
            "margin": np.abs(np.asarray(logit_hate, dtype=np.float64) - np.asarray(logit_nonhate, dtype=np.float64)),
            "entropy": entropy,
        }
    )
    for key, values in (extra or {}).items():
        # Scalars (e.g. model_id="") must broadcast; only sequences are per-row.
        if values is None or isinstance(values, (str, bytes, int, float, bool)):
            frame[key] = values
        else:
            column = list(values)
            if len(column) != len(frame):
                raise ValueError(
                    f"extra[{key!r}] has length {len(column)} but the frame has {len(frame)} rows"
                )
            frame[key] = column
    return frame


def extract_logits_hf(
    model: Any,
    tokenizer: Any,
    dataset: pd.DataFrame,
    verbalizer: dict[str, str] | None = None,
    batch_size: int = 16,
    *,
    split: str = "eval",
    max_len: int = 256,
    template: str = DEFAULT_PROMPT_TEMPLATE,
    model_id: str = "",
    device: str | torch.device | None = None,
    progress_every: int = 20,
) -> pd.DataFrame:
    """Extract two-way label logits for every row of ``dataset``.

    ``dataset`` needs ``text``, ``label`` and (for provenance) ``example_id``.
    Returns the canonical logits frame with columns ``REQUIRED_COLUMNS`` plus
    ``correct``/``margin``/``entropy``.
    """
    v = verbalizer or DEFAULT_VERBALIZER
    hate_token_id = encode_label_token(tokenizer, v["hate"])
    nonhate_token_id = encode_label_token(tokenizer, v["nonhate"])
    if hate_token_id == nonhate_token_id:
        raise ValueError(f"Verbalizer tokens collide on id {hate_token_id}")

    if device is not None:
        model = model.to(device)
    was_training = bool(getattr(model, "training", False))
    model.eval()
    if hasattr(model, "use_cache"):
        model.config.use_cache = False

    examples = build_eval_examples(dataset, template)
    prompts = [e["prompt"] for e in examples]
    ids = [e["example_id"] for e in examples]
    labels = [e["y_true"] for e in examples]
    domains = [e["domain"] for e in examples]

    hate_chunks: list[np.ndarray] = []
    nonhate_chunks: list[np.ndarray] = []
    t0 = time.perf_counter()
    n_batches = math_ceil(len(prompts), batch_size)
    for i, start in enumerate(range(0, len(prompts), batch_size)):
        chunk = prompts[start : start + batch_size]
        lh, ln = answer_logits_batched(
            model, tokenizer, chunk, hate_token_id, nonhate_token_id, max_len=max_len
        )
        hate_chunks.append(lh)
        nonhate_chunks.append(ln)
        if progress_every and (i + 1) % progress_every == 0:
            LOG.info(
                "logit extraction %d/%d batches (%.1fs)", i + 1, n_batches, time.perf_counter() - t0
            )

    frame = probs_to_frame(
        ids,
        split,
        labels,
        np.concatenate(hate_chunks) if hate_chunks else np.zeros(0),
        np.concatenate(nonhate_chunks) if nonhate_chunks else np.zeros(0),
        extra={"domain": domains, "model": model_id, "language": dataset.get("language", pd.Series(["bn"] * len(dataset))).tolist()},
    )
    if was_training and hasattr(model, "train"):
        model.train()
    LOG.info("Extracted %d rows in %.1fs", len(frame), time.perf_counter() - t0)
    return frame


def math_ceil(a: int, b: int) -> int:
    return -(-a // b)


def validate_logits_frame(df: pd.DataFrame, prob_tol: float = 1e-9) -> dict[str, Any]:
    """T3.5/T3.10 gate: columns present, rows complete, probabilities normalised."""
    missing = [c for c in REQUIRED_COLUMNS if c not in df.columns]
    if missing:
        raise ValueError(f"Missing required logits columns: {missing}")
    if len(df) == 0:
        raise ValueError("Empty logits frame")
    sums = df["prob_hate"].to_numpy() + df["prob_nonhate"].to_numpy()
    max_dev = float(np.max(np.abs(sums - 1.0)))
    if max_dev > prob_tol:
        raise ValueError(f"prob_hate + prob_nonhate deviates from 1 by {max_dev}")
    if df["example_id"].duplicated().any():
        raise ValueError("Duplicate example_id in logits frame")
    if not np.isfinite(df[_LOGIT_COLUMNS].to_numpy(dtype=np.float64)).all():
        raise ValueError("Non-finite values in logits frame")
    return {
        "n_rows": int(len(df)),
        "columns": list(df.columns),
        "missing_columns": missing,
        "max_prob_sum_deviation": max_dev,
        "finite": True,
        "unique_example_ids": True,
    }


def save_logits(df: pd.DataFrame, path: str | Path, **meta: Any) -> Path:
    """Persist per-example logits as Parquet (recomputable metrics later)."""
    out = Path(path)
    out.parent.mkdir(parents=True, exist_ok=True)
    payload = df.copy()
    for k, v in meta.items():
        payload[k] = v
    payload.to_parquet(out, index=False)
    LOG.info("Saved logits -> %s (%.2f MB)", out, out.stat().st_size / 1e6)
    return out


def load_logits(path: str | Path) -> pd.DataFrame:
    return pd.read_parquet(path)


def agreement_between_frames(a: pd.DataFrame, b: pd.DataFrame, id_col: str = "example_id") -> dict[str, Any]:
    """Prediction agreement between two logits frames aligned on ``id_col``."""
    left = a.set_index(id_col)
    right = b.set_index(id_col).reindex(left.index)
    if right.isna()["prediction"].any():
        missing = int(right["prediction"].isna().sum())
        raise ValueError(f"{missing} example_ids missing from the second logits frame")
    same = (left["prediction"].to_numpy() == right["prediction"].to_numpy())
    return {
        "n": int(len(left)),
        "agreement_rate": float(same.mean()),
        "max_abs_prob_delta": float(
            np.max(np.abs(left["prob_hate"].to_numpy() - right["prob_hate"].to_numpy()))
        ),
        "mean_abs_prob_delta": float(
            np.mean(np.abs(left["prob_hate"].to_numpy() - right["prob_hate"].to_numpy()))
        ),
    }
```

```python
"""Track A post-training quantization (GPU / Hugging Face).

Track A never pools with Track B (global rule 4).  Every method here returns a
model that the shared :func:`src.models.logit_extraction.extract_logits_hf` can
read, so downstream metrics are identical across precision settings.

GPTQ calibration data must come **only** from the training split (global rule 5).
"""

from __future__ import annotations

import gc
import logging
import time
from pathlib import Path
from typing import Any

import pandas as pd
import torch

from src.models.finetune import DEFAULT_PROMPT_TEMPLATE, build_prompt

TRACK_A_METHODS = ("bnb_int8", "bnb_nf4", "gptq_int4", "awq_int4")


def bnb_int8_config() -> Any:
    from transformers import BitsAndBytesConfig

    return BitsAndBytesConfig(load_in_8bit=True)


def bnb_nf4_config(compute_dtype: str = "bfloat16") -> Any:
    from transformers import BitsAndBytesConfig

    dtype = {"bfloat16": torch.bfloat16, "float16": torch.float16, "float32": torch.float32}[
        compute_dtype
    ]
    return BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_quant_type="nf4",
        bnb_4bit_compute_dtype=dtype,
        bnb_4bit_use_double_quant=True,
    )


def load_quantized_hf(model_path: str, method: str, compute_dtype: str = "bfloat16"):
    """Load ``model_path`` under the requested on-the-fly Track A method."""
    from src.models.finetune import load_tokenizer  # noqa: F401  (import guard)
    from transformers import AutoModelForCausalLM

    if method == "bnb_int8":
        qc = bnb_int8_config()
    elif method == "bnb_nf4":
        qc = bnb_nf4_config(compute_dtype)
    else:
        raise ValueError(f"{method!r} is not a load-time bnb method; quantize it first")

    model = AutoModelForCausalLM.from_pretrained(
        model_path,
        quantization_config=qc,
        device_map="auto",
        # transformers <4.56 spells this `torch_dtype`; `dtype` was added later.
        torch_dtype=torch.bfloat16 if compute_dtype == "bfloat16" else torch.float16,
    )
    model.config.use_cache = False
    model.eval()
    return model


def free(model: Any) -> None:
    """Drop a model and reclaim VRAM before loading the next precision."""
    if model is not None:
        del model
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
        torch.cuda.synchronize()

def build_calibration_texts(
    train_df: pd.DataFrame,
    n_samples: int = 128,
    max_len: int = 256,
    seed: int = 42,
    template: str = DEFAULT_PROMPT_TEMPLATE,
) -> list[str]:
    """Calibration prompts drawn **only** from the training split."""
    if n_samples >= len(train_df):
        picked = train_df
    else:
        picked = train_df.sample(n=n_samples, random_state=seed)
    return [build_prompt(t, template) for t in picked["text"].tolist()]


class GPTQUnavailable(RuntimeError):
    """The installed gptqmodel build exposes no end-to-end quantization driver."""


def gptq_backend_status() -> dict[str, Any]:
    """Report which GPTQ driver APIs the installed ``gptqmodel`` build actually offers.

    gptqmodel restructured across 1.x -> 2.x. Older builds expose the driver as
    ``GPTQModel.load(...).quantize(tokenizer, calib)`` then ``.save(dir)``; newer
    builds ship only the per-layer :class:`gptqmodel.quantization.gptq.GPTQ`
    accumulator (``add_batch`` / ``fasterquant``) and expect the caller to write
    the layer-sequential loop, which we deliberately do not do.
    """
    status: dict[str, Any] = {
        "installed": False,
        "version": None,
        "driver_api": None,
        "has_model_quantize": False,
        "has_model_save": False,
        "low_level_api": [],
        "problems": [],
    }
    try:
        import gptqmodel
        from gptqmodel import GPTQModel
    except Exception as exc:
        status["problems"].append(f"import failed: {exc!r}")
        return status

    status["installed"] = True
    status["version"] = getattr(gptqmodel, "__version__", "unknown")
    status["has_model_quantize"] = callable(getattr(GPTQModel, "quantize", None))
    status["has_model_save"] = callable(getattr(GPTQModel, "save", None))
    if status["has_model_quantize"] and status["has_model_save"]:
        status["driver_api"] = "GPTQModel.load/.quantize/.save"
        return status

    try:
        from gptqmodel.quantization.gptq import GPTQ

        status["low_level_api"] = [n for n in ("add_batch", "fasterquant") if hasattr(GPTQ, n)]
    except Exception as exc:
        status["problems"].append(f"low-level GPTQ import failed: {exc!r}")

    status["problems"].append(
        f"gptqmodel {status['version']} exposes no end-to-end driver "
        f"(GPTQModel.quantize={status['has_model_quantize']}, "
        f"GPTQModel.save={status['has_model_save']}); only the low-level per-layer "
        f"{status['low_level_api']} API is present and no `gptqmodel` CLI is installed, "
        "so GPTQ-4 cannot be produced without a hand-written sequential driver."
    )
    return status


def quantize_gptq(
    model_path: str,
    calib_texts: list[str],
    output_path: str,
    *,
    bits: int = 4,
    group_size: int = 128,
    tokenizer_path: str | None = None,
    batch_size: int = 1,
    max_len: int = 256,
    device: str = "cuda:0",
    trust_remote_code: bool = False,
) -> dict[str, Any]:
    """GPTQ weight-only quantization via GPTQModel.

    ``calib_texts`` must be sampled from the training split only.

    Raises:
        GPTQUnavailable: if the installed build has no ``load -> quantize -> save``
            driver (see :func:`gptq_backend_status`).
    """
    status = gptq_backend_status()
    if status["driver_api"] is None:
        raise GPTQUnavailable("; ".join(status["problems"]) or "gptqmodel driver API not found")

    from gptqmodel import GPTQModel, QuantizeConfig
    from transformers import AutoTokenizer

    if not calib_texts:
        raise ValueError("GPTQ requires a non-empty calibration set from the training split")

    tok_path = tokenizer_path or model_path
    tokenizer = AutoTokenizer.from_pretrained(tok_path)
    if tokenizer.pad_token is None:
        tokenizer.pad_token = tokenizer.eos_token
    tokenizer.padding_side = "left"

    dynamic_enum = None
    for candidate in ("DYNAMIC",):
        try:
            from gptqmodel import DYNAMIC  # type: ignore

            dynamic_enum = getattr(DYNAMIC, "AUTO", None)
        except Exception:
            try:
                from gptqmodel.utils import DYNAMIC  # type: ignore

                dynamic_enum = getattr(DYNAMIC, "AUTO", None)
            except Exception:
                dynamic_enum = None
    quant_kwargs: dict[str, Any] = {"bits": bits, "group_size": group_size, "desc_act": True}
    if dynamic_enum is not None:
        quant_kwargs["dynamic"] = dynamic_enum
    quant_config = QuantizeConfig(**quant_kwargs)

    t0 = time.perf_counter()
    model = GPTQModel.load(model_path, device=device, trust_remote_code=trust_remote_code)
    load_s = time.perf_counter() - t0

    t1 = time.perf_counter()
    with torch.no_grad():
        model.quantize(tokenizer, calib_texts, batch_size=batch_size, max_len=max_len)
    quantize_s = time.perf_counter() - t1

    Path(output_path).mkdir(parents=True, exist_ok=True)
    t2 = time.perf_counter()
    model.save(str(output_path), safe_serialization=True)
    tokenizer.save_pretrained(str(output_path))
    save_s = time.perf_counter() - t2

    try:
        del model
        free(None)
    except Exception:
        pass

    return {
        "method": "gptq_int4",
        "bits": bits,
        "group_size": group_size,
        "desc_act": True,
        "dynamic": dynamic_enum is not None,
        "n_calib": len(calib_texts),
        "output_path": str(output_path),
        "load_s": load_s,
        "quantize_s": quantize_s,
        "save_s": save_s,
        "total_s": load_s + quantize_s + save_s,
        "backend": status,
    }


def load_gptq(model_path: str, device: str = "cuda:0"):
    """Reload a saved GPTQ checkpoint through the standard HF auto classes."""
    from transformers import AutoModelForCausalLM, AutoTokenizer

    tokenizer = AutoTokenizer.from_pretrained(model_path)
    if tokenizer.pad_token is None:
        tokenizer.pad_token = tokenizer.eos_token
    tokenizer.padding_side = "right"
    model = AutoModelForCausalLM.from_pretrained(
        model_path, device_map="auto", torch_dtype=torch.bfloat16
    )
    model.eval()
    model.config.use_cache = False
    return model, tokenizer


def quantize_awq(
    model_path: str,
    calib_texts: list[str],
    output_path: str,
    *,
    bits: int = 4,
    group_size: int = 128,
    max_len: int = 256,
    device: str = "cuda:0",
) -> dict[str, Any]:
    """AWQ (activation-aware) weight-only INT4.

    AWQ ships with ``llmcompressor`` rather than the GPTQ stack pinned here; this
    wrapper keeps the pilot's method table complete and reports availability
    honestly instead of silently substituting GPTQ.
    """
    try:
        from llmcompressor.transformers import oneshot  # noqa: F401
    except ImportError as exc:
        raise RuntimeError(
            "AWQ requires llmcompressor, which is not installed in this environment. "
            "Install it (Phase 6) or drop awq_int4 from the pilot method list."
        ) from exc
    raise NotImplementedError("AWQ path lands in Phase 6; Phase 3 pilots GPTQ-4 only.")


def quantize_track_a(
    model_path: str,
    method: str,
    output_path: str | None = None,
    *,
    calib_texts: list[str] | None = None,
    bits: int = 4,
    group_size: int = 128,
    max_len: int = 256,
    device: str = "cuda:0",
) -> dict[str, Any]:
    """Dispatch to the requested Track A method."""
    if method not in TRACK_A_METHODS:
        raise ValueError(f"Unknown Track A method {method!r}; expected one of {TRACK_A_METHODS}")
    if method in {"bnb_int8", "bnb_nf4"}:
        return {"method": method, "mode": "on_load", "model_path": model_path}
    if method == "gptq_int4":
        if output_path is None:
            raise ValueError("gptq_int4 needs an output_path")
        if calib_texts is None:
            raise ValueError("gptq_int4 needs calibration texts from the training split")
        return quantize_gptq(
            model_path,
            calib_texts,
            output_path,
            bits=bits,
            group_size=group_size,
            max_len=max_len,
            device=device,
        )
    if method == "awq_int4":
        if calib_texts is None:
            raise ValueError("awq_int4 needs calibration texts from the training split")
        return quantize_awq(
            model_path, calib_texts, output_path or "", bits=bits, group_size=group_size, max_len=max_len
        )
    raise AssertionError(f"unhandled method {method!r}")


def artifact_size_mb(path: str | Path) -> float:
    """Total size of a saved model directory in MB."""
    p = Path(path)
    if p.is_file():
        return p.stat().st_size / 1e6
    return sum(f.stat().st_size for f in p.rglob("*") if f.is_file()) / 1e6


def quant_summary(model: Any) -> dict[str, Any]:
    """Parameter/quantisation summary of a live model."""
    from src.metrics.efficiency import model_size_bytes, peak_vram_gb

    try:
        from transformers.quantizers.quantizer_bnb_4bit import (
            Params4bit,
        )  # noqa: F401
    except Exception:
        pass
    n_params = sum(p.numel() for p in model.parameters())
    n_4bit = n_8bit = 0
    for p in model.parameters():
        cls = type(p).__name__
        if "Int8" in cls or "int8" in cls.lower():
            n_8bit += p.numel()
        elif "4bit" in cls.lower():
            n_4bit += p.numel()
    return {
        "n_parameters": int(n_params),
        "n_params_4bit": int(n_4bit),
        "n_params_8bit": int(n_8bit),
        "weight_bytes": model_size_bytes(model),
        "peak_vram_gib": None if peak_vram_gb() is None else round(peak_vram_gb() or 0.0, 3),
        "class": type(model).__name__,
    }


def build_gptq_calib_from_train(train_df: pd.DataFrame, n_samples: int, max_len: int, seed: int) -> list[str]:
    """Convenience wrapper enforcing the train-split-only calibration rule."""

def calibration_provenance(train_df: pd.DataFrame, calib_texts: list[str], forbidden_ids: set[str]) -> dict[str, Any]:
    """Assert the calibration prompts never touch id_val/id_test/ood examples."""
    sample = calib_texts[: min(50, len(calib_texts))]
    joined = "\n".join(sample)
    leaked = [eid for eid in forbidden_ids if eid and eid in joined]
    return {
        "n_calib": len(calib_texts),
        "forbidden_ids_checked": len(forbidden_ids),
        "leaked_ids": leaked,
        "ok": not leaked,
        "source": "train_split_only",
    }


def calib_texts_from_frame(frame: pd.DataFrame) -> pd.DataFrame:
    """Sanity accessor used by the pilot script's manifest."""
    if frame.empty:
        raise ValueError("Empty training frame for GPTQ calibration")
    return frame[["example_id", "text", "label"]].head(1)
```

```python
def _perfect_ranker(n_errors: int = 1, n: int = 100):
    """Errors placed at the lowest confidence -> optimal risk-coverage ranking.

    Every example is predicted positive (``y_prob >= 0.5``); the ``n_errors``
    negatives get the smallest confidence so they are accepted last.
    """
    n_errors = min(n_errors, n)
    y_true = np.ones(n, dtype=int)
    y_prob = np.full(n, 0.9)
    y_true[:n_errors] = 0
    y_prob[:n_errors] = 0.51
    return y_true, y_prob


def _worst_ranker(n_errors: int = 1, n: int = 100):
    """Same error count, but errors carry the *highest* confidence."""
    n_errors = min(n_errors, n)
    y_true = np.ones(n, dtype=int)
    y_prob = np.full(n, 0.6)
    y_true[:n_errors] = 0
    y_prob[:n_errors] = 0.99
    return y_true, y_prob


class TestRiskCoverageCurve:
    def test_shapes_and_monotonic_coverage(self):

        assert cov.shape == risk.shape == (200,)
        assert cov[0] == pytest.approx(1 / 200)
        assert cov[-1] == pytest.approx(1.0)
        assert np.all(np.diff(cov) > 0)
```

```python
"""Tests for per-example logit extraction (Phase 3, T3.5 / T3.10).

The central regression test is :func:`test_batch_composition_does_not_change_logits`.
Reading ``logits[:, -1, :]`` on a right-padded batch silently scores the pad slot
of every row shorter than the longest prompt, so a short example gets a different
answer depending on what it was batched with.
"""

from __future__ import annotations

from types import SimpleNamespace

import numpy as np
import pandas as pd
import pytest
import torch

from src.models import logit_extraction as lx

HATE_ID = 7
NONHATE_ID = 9
VERBALIZER_IDS = {" A": HATE_ID, " B": NONHATE_ID}
VERBALIZER = {"hate": " A", "nonhate": " B"}


class FakeTokenizer:
    """Minimal right-padding tokenizer that honours ``truncation_side``."""

    pad_token_id = 0

    def __init__(self, truncation_side: str = "right") -> None:
        self.truncation_side = truncation_side
        self.seen_truncation_side: str | None = None
        self.n_calls = 0

    def encode(self, text: str, add_special_tokens: bool = False) -> list[int]:
        if text in VERBALIZER_IDS:
            return [VERBALIZER_IDS[text]]
        return [ord(c) % 20 + 1 for c in text]

    def __call__(
        self,
        texts,
        add_special_tokens: bool = False,
        padding: bool = True,
        truncation: bool = False,
        max_length: int | None = None,
        return_tensors: str | None = None,
    ) -> dict[str, torch.Tensor]:
        self.n_calls += 1
        self.seen_truncation_side = self.truncation_side
        if isinstance(texts, str):
            texts = [texts]
        seqs = [self.encode(t) for t in texts]
        if truncation and max_length is not None:
            if self.truncation_side == "left":
                seqs = [s[-max_length:] for s in seqs]
            else:
                seqs = [s[:max_length] for s in seqs]
        width = max(len(s) for s in seqs)
        ids = torch.zeros(len(seqs), width, dtype=torch.long)
        mask = torch.zeros(len(seqs), width, dtype=torch.long)
        for i, seq in enumerate(seqs):
            ids[i, : len(seq)] = torch.tensor(seq, dtype=torch.long)
            mask[i, : len(seq)] = 1
        return {"input_ids": ids, "attention_mask": mask}


class PositionEchoModel(torch.nn.Module):
    """``logits[b, t, v] = t + 0.001 * v``.

    Reading the correct answer position is therefore observable directly: the
    returned value must equal ``(last_real_index + 0.001 * token_id)``.
    """

    def __init__(self, vocab_size: int = 32) -> None:
        super().__init__()
        self.vocab_size = vocab_size
        self.anchor = torch.nn.Parameter(torch.zeros(1))

    def forward(self, input_ids=None, attention_mask=None, use_cache=None, **kwargs):
        b, t = input_ids.shape
        pos = torch.arange(t, dtype=torch.float32).view(1, t, 1)
        vid = torch.arange(self.vocab_size, dtype=torch.float32).view(1, 1, -1)
        return SimpleNamespace(logits=(pos + 0.001 * vid).expand(b, t, self.vocab_size).clone())

def test_batch_composition_does_not_change_logits():
```

```python
class TestBrier:
    def test_perfect_predictions(self):

        y_prob = rng.uniform(0, 1, 300)

        rng = np.random.default_rng(4)

        assert abs(ours - theirs) < 0.02, f"ours={ours} netcal={theirs}"
        assert ours == pytest.approx(theirs, abs=1e-9)

    def test_mce_matches_netcal(self):
        rng = np.random.default_rng(5)
        y_true = rng.integers(0, 2, 1000)
        y_prob = rng.uniform(0, 1, 1000)
        ours = maximum_calibration_error(y_true, y_prob, n_bins=15, strategy="equal_width")
        theirs = self._measure(MCE, y_true, y_prob, 15, equal_intervals=True)
        assert abs(ours - theirs) < 1e-9, f"ours={ours} netcal={theirs}"

    def test_ece_mass_matches_netcal_ace(self):
        """Adaptive (equal-mass) ECE agrees with netcal's ACE within 0.02."""
        rng = np.random.default_rng(8)
        y_true = rng.integers(0, 2, 2000)
        y_prob = rng.uniform(0, 1, 2000)
        ours = expected_calibration_error(y_true, y_prob, n_bins=15, strategy="equal_mass")
        theirs = self._measure(ACE, y_true, y_prob, 15)
        assert abs(ours - theirs) < 0.02, f"ours={ours} netcal={theirs}"

    def test_ece_matches_netcal_on_overconfident(self):
        """Sanity: agreement also holds in the skewed regime the study cares about."""
        rng = np.random.default_rng(6)
        y_true = rng.integers(0, 2, 1200)
        y_prob = np.clip(rng.beta(2.0, 2.0, 1200) ** 4, 0.0, 1.0)
        ours = expected_calibration_error(y_true, y_prob, n_bins=15, strategy="equal_width")
        theirs = self._measure(ECE, y_true, y_prob, 15, equal_intervals=True)
        assert abs(ours - theirs) < 0.02, f"ours={ours} netcal={theirs}"

    def test_ece_matches_netcal_on_ood_prior_shift(self):
        """OOD folds have a shifted class prior; agreement must survive that."""

        y_true = (rng.uniform(0, 1, 1500) < 0.95).astype(int)  # heavily imbalanced
        y_prob = np.clip(rng.beta(1.5, 6.0, 1500), 0.0, 1.0)
        ours = expected_calibration_error(y_true, y_prob, n_bins=15, strategy="equal_width")
```

```python
    """Best-effort parse of a version string into an int tuple."""
    parts: list[int] = []
    for chunk in version_str.split("+")[0].split("."):
        digits = ""
        for ch in chunk:
            if ch.isdigit():
                digits += ch
            else:
                break
        if digits:
            parts.append(int(digits))
        else:
            break
    return tuple(parts)

    try:

    return info

def run_pip_check() -> dict[str, Any]:
    """Run `pip check` and return structured result.

    With --system-site-packages, pip surfaces conflicts from unrelated global
    packages (opencv, gradio, anylabeling, ...). T0.2 only fails when a
    conflict is *internal* to this project: both the complaining package and
    the package whose version is wrong are project-owned.
    """
    result = subprocess.run(
        [sys.executable, "-m", "pip", "check"],
        capture_output=True,
        text=True,
        check=False,
    )
    output = ((result.stdout or "") + (result.stderr or "")).strip()
    lines = [ln.strip() for ln in output.splitlines() if ln.strip()]

    import re

    pattern = re.compile(
        r"^(?P<complainer>\S+) has requirement (?P<req>\S+), but you have (?P<have>\S+)"
    )
    project_conflicts: list[str] = []
    for ln in lines:
        m = pattern.match(ln)
        if not m:
            continue
        complainer = m.group("complainer").split()[0]
        have = m.group("have").split()[0]
        # Internal project conflict: project package complaining about another
        # project package's version.
        if complainer in PROJECT_PACKAGES and have in PROJECT_PACKAGES:
            project_conflicts.append(ln)

    passed = len(project_conflicts) == 0
    LOG.info(
        "pip check project_conflicts=%s (total_lines=%s)\n%s",
        len(project_conflicts),
        len(lines),
        output,
    )
    return {
        "passed": passed,
        "returncode": result.returncode,
        "project_conflicts": project_conflicts,
        "all_conflicts": lines,
        "output": output,
    }

def pin_versions() -> dict[str, str]:
    """Generate version manifest and save outputs/environment_versions.json."""
    versions: dict[str, str] = {}
    for lib in VERSION_LIBS:
        try:
            mod = importlib.import_module(lib)
            versions[lib] = getattr(mod, "__version__", "unknown")
        except Exception as exc:  # noqa: BLE001 — record any import failure
            versions[lib] = f"NOT INSTALLED ({type(exc).__name__})"

    # Provider notes for deprecated packages.
    notes: dict[str, str] = {}
    if versions.get("gptqmodel", "").startswith("NOT"):
        notes["gptq"] = "GPTQModel not installed"
    else:
        notes["gptq"] = "GPTQModel installed (maintained AutoGPTQ/AutoAWQ successor)"
    if versions.get("auto_gptq", "").startswith("NOT"):
        notes["auto_gptq"] = "legacy AutoGPTQ not installed; using GPTQModel"
    if versions.get("awq", "").startswith("NOT"):
        notes["awq"] = "legacy AutoAWQ not installed; AWQ available via GPTQModel/transformers"

    manifest = {
        "generated_at_utc": _utc_now(),
        "python": platform.python_version(),
        "platform": platform.platform(),
        "versions": versions,
        "notes": notes,
    }
    out = ROOT / "outputs" / "environment_versions.json"
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(manifest, indent=2), encoding="utf-8")
    LOG.info("Wrote %s", out)
    return versions


def _required_libs_satisfied(versions: dict[str, str]) -> tuple[bool, list[str]]:
    """Core libs that must import for the pipeline to run."""
    core = [
        "torch",
        "transformers",
        "peft",
        "accelerate",
        "bitsandbytes",
        "datasets",
        "scipy",
        "statsmodels",
        "numpy",
        "pandas",
        "sklearn",
        "pynvml",
        "llama_cpp",
    ]
    missing = [
        name
        for name in core
        if versions.get(name, "").startswith("NOT")
    ]
    # GPTQ backend: at least one of gptqmodel / auto_gptq must be present.
    gptq_ok = not (
        versions.get("gptqmodel", "").startswith("NOT")
        and versions.get("auto_gptq", "").startswith("NOT")
    )
    if not gptq_ok:
        missing.append("gptqmodel|auto_gptq")
    return (not missing, missing)

def _ensure_tiny_model(cache_dir: Path) -> str:
    """Return a local path for a tiny causal LM usable in smoke tests.

    Synthetic GPT-2 with n_embd=64 so GPTQ group_size=32 is valid. Embeddings
    are resized to the tokenizer vocab so generate() cannot OOB. config.json
    also stores n_ctx (llama.cpp convert expects it).
    """
    import json

    import torch
    from transformers import AutoTokenizer, GPT2Config, GPT2LMHeadModel

    local = cache_dir / "tiny_smoke_gpt2"
    if (local / "config.json").exists():
        return str(local)

    config = GPT2Config(
        vocab_size=1000,
        n_positions=128,
        n_embd=64,
        n_layer=2,
        n_head=4,
        bos_token_id=0,
        eos_token_id=0,
        # Untied lm_head so HF safetensors exports lm_head.weight and
        # llama.cpp emits output.weight (required by llama-cpp-python 0.3.4).
        tie_word_embeddings=False,
    )
    model = GPT2LMHeadModel(config)
    model.lm_head.weight = torch.nn.Parameter(model.lm_head.weight.detach().clone())
    try:
        tokenizer = AutoTokenizer.from_pretrained("gpt2")
        if tokenizer.pad_token is None:
            tokenizer.pad_token = tokenizer.eos_token
    except Exception:  # noqa: BLE001 — offline / hub issues
        tokenizer = AutoTokenizer.from_pretrained(TINY_MODEL_ID)
        if tokenizer.pad_token is None:
            tokenizer.pad_token = tokenizer.eos_token or tokenizer.unk_token
    model.resize_token_embeddings(len(tokenizer))
    config.vocab_size = len(tokenizer)
    local.mkdir(parents=True, exist_ok=True)
    model.save_pretrained(local)
    tokenizer.save_pretrained(local)
    # llama.cpp GPT-2 converter reads hparams["n_ctx"].
    cfg_path = local / "config.json"
    cfg = json.loads(cfg_path.read_text(encoding="utf-8"))
    cfg["n_ctx"] = cfg.get("n_positions", 128)
    cfg_path.write_text(json.dumps(cfg, indent=2), encoding="utf-8")
    LOG.info("Built local tiny GPT-2 at %s (vocab=%s)", local, cfg.get("vocab_size"))
    return str(local)


def smoke_test_bnb(model_path: str) -> dict[str, Any]:
    """Load a tiny model in INT8 and NF4 to verify bitsandbytes works."""
    import torch
    from transformers import AutoModelForCausalLM, BitsAndBytesConfig

    results: dict[str, Any] = {"model_path": model_path}

    # INT8
    config_8 = BitsAndBytesConfig(load_in_8bit=True)
    model_8 = AutoModelForCausalLM.from_pretrained(
        model_path,
        quantization_config=config_8,
        device_map="auto",
        torch_dtype=torch.float16,
    )
    assert model_8 is not None
    with torch.no_grad():
        dummy = torch.tensor([[0, 0]], device=model_8.device)
        out_8 = model_8(dummy)
    assert out_8.logits is not None and out_8.logits.numel() > 0
    del model_8
    torch.cuda.empty_cache()
    results["int8"] = "ok"

    # NF4
    config_4 = BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_quant_type="nf4",
        bnb_4bit_compute_dtype=torch.float16,
    )
    model_4 = AutoModelForCausalLM.from_pretrained(
        model_path,
        quantization_config=config_4,
        device_map="auto",
        torch_dtype=torch.float16,
    )
    assert model_4 is not None
    with torch.no_grad():
        dummy = torch.tensor([[0, 0]], device=model_4.device)
        out_4 = model_4(dummy)
    assert out_4.logits is not None and out_4.logits.numel() > 0
    del model_4
    torch.cuda.empty_cache()
    results["nf4"] = "ok"

    LOG.info("bitsandbytes INT8 and NF4 smoke test passed")
    return results

def smoke_test_gptq(model_path: str) -> dict[str, Any]:
    """Verify GPTQ quantization works end-to-end on a tiny model."""
    from transformers import AutoTokenizer

    tokenizer = AutoTokenizer.from_pretrained(model_path)
    if tokenizer.pad_token is None:
        tokenizer.pad_token = tokenizer.eos_token or tokenizer.unk_token

    calib_texts = [
        "The quick brown fox jumps over the lazy dog.",
        "Quantization changes model confidence under domain shift.",
        "Bengali hate-speech classification requires careful calibration.",
        "Selective prediction abstains when confidence is low.",
        "Temperature scaling fits a single scalar on validation logits.",
        "INT4 weights trade accuracy for edge-device efficiency.",
        "Calibration error is measured with ECE and Brier score.",
        "Risk-coverage curves summarize selective prediction quality.",
        "Domain shift degrades both accuracy and calibration.",
        "Small language models still need trustworthy confidence.",
    ]
    calib_data = []
    for text in calib_texts:
        enc = tokenizer(
            text,
            return_tensors="pt",
            truncation=True,
            max_length=64,
        )
        calib_data.append(
            {
                "input_ids": enc["input_ids"],
                "attention_mask": enc["attention_mask"],
            }
        )

    out_dir = ROOT / "outputs" / "quantized" / "smoke_gptq"
    out_dir.mkdir(parents=True, exist_ok=True)

    # Prefer maintained GPTQModel; fall back to legacy auto_gptq.
    backend_used = None
    last_error: Exception | None = None

    try:
        from gptqmodel import GPTQModel, QuantizeConfig

        # Tiny models often need a small group_size (dims may be < 128).
        group_size = 32
        quant_config = QuantizeConfig(bits=4, group_size=group_size, desc_act=False)
        model = GPTQModel.load(model_path, quantize_config=quant_config)
        if not hasattr(model, "quantize"):
            raise RuntimeError("GPTQModel instance has no quantize() method")
        model.quantize(calib_data)
        if hasattr(model, "save"):
            model.save(str(out_dir))
        elif hasattr(model, "save_quantized"):
            model.save_quantized(str(out_dir))
        else:
            model.save_pretrained(str(out_dir))
        reloaded = GPTQModel.load(str(out_dir))
        emb = reloaded.get_input_embeddings()
        vocab = int(emb.weight.shape[0]) if emb is not None else len(tokenizer)
        prompt = tokenizer("Hello", return_tensors="pt")
        input_ids = prompt["input_ids"].to(reloaded.device).clamp(max=vocab - 1)
        attention_mask = prompt.get("attention_mask")
        if attention_mask is not None:
            attention_mask = attention_mask.to(reloaded.device)
        result = None
        if hasattr(reloaded, "generate"):
            try:
                result = reloaded.generate(
                    input_ids=input_ids,
                    attention_mask=attention_mask,
                    max_new_tokens=4,
                )
            except Exception as gen_exc:  # noqa: BLE001
                LOG.warning("GPTQ generate failed (%s); falling back to forward", gen_exc)
        if result is None:
            outputs = reloaded(input_ids=input_ids, attention_mask=attention_mask)
            result = outputs.logits if hasattr(outputs, "logits") else outputs
        assert result is not None
        backend_used = "gptqmodel"
        del model, reloaded
    except Exception as exc:  # noqa: BLE001
        last_error = exc
        LOG.warning("GPTQModel smoke failed (%s); trying auto_gptq", exc)

    if backend_used is None:
        try:
            from auto_gptq import AutoGPTQForCausalLM, BaseQuantizeConfig

            qconfig = BaseQuantizeConfig(bits=4, group_size=32, desc_act=False)
            model = AutoGPTQForCausalLM.from_pretrained(model_path, quantize_config=qconfig)
            model.quantize(calib_data)
            model.save_quantized(str(out_dir))
            model.save_quantized(str(out_dir), safetensors=True)
            reloaded = AutoGPTQForCausalLM.from_pretrained(
                str(out_dir), quantize_config=qconfig
            )
            prompt = tokenizer("Hello", return_tensors="pt")
            outputs = reloaded(**prompt)
            assert outputs.logits is not None and outputs.logits.numel() > 0
            backend_used = "auto_gptq"
            del model, reloaded
        except Exception as exc:  # noqa: BLE001
            last_error = exc

    if backend_used is None:
        raise RuntimeError(f"GPTQ smoke test failed on both backends: {last_error}")

    LOG.info("GPTQ smoke test passed via %s", backend_used)
    return {"backend": backend_used, "output_dir": str(out_dir)}

def _convert_hf_to_gguf(model_path: str, gguf_path: Path) -> None:
    """Convert a local HF model directory to GGUF using llama.cpp's converter."""
    gguf_path.parent.mkdir(parents=True, exist_ok=True)

    # Prefer a local llama.cpp checkout (has conversion/ + gguf-py/).
    local_llama = ROOT / "outputs" / "quantized" / "llama.cpp"
    convert_script = local_llama / "convert_hf_to_gguf.py"
    if not convert_script.exists():
        convert_script = _download_convert_script(
            ROOT / "outputs" / "quantized" / "convert_hf_to_gguf.py"
        )
        run_cwd = convert_script.parent
        python_path_parts = [str(run_cwd), str(run_cwd / "gguf-py")]
    else:
        run_cwd = local_llama
        python_path_parts = [str(local_llama), str(local_llama / "gguf-py")]

    cmd = [
        sys.executable,
        str(convert_script),
        model_path,
        "--outfile",
        str(gguf_path),
        "--outtype",
        "f32",
    ]
    LOG.info("Running: %s (cwd=%s)", " ".join(cmd), run_cwd)
    env = os.environ.copy()
    env["PYTHONUTF8"] = "1"
    env["PYTHONIOENCODING"] = "utf-8"
    env["PYTHONPATH"] = os.pathsep.join(
        python_path_parts + [env.get("PYTHONPATH", "")]
    ).rstrip(os.pathsep)
    result = subprocess.run(
        cmd,
        capture_output=True,
        text=True,
        check=False,
        env=env,
        cwd=str(run_cwd),
    )
    if result.returncode != 0 or not gguf_path.exists():
        raise RuntimeError(
            "convert_hf_to_gguf.py failed:\n"
            f"stdout:\n{result.stdout[-4000:]}\n"
            f"stderr:\n{result.stderr[-4000:]}"
        )


def _download_tiny_llamas_gguf(dest_dir: Path) -> Path:
    """Download a tiny official GGUF (fallback path for llama.cpp inference)."""
    from huggingface_hub import hf_hub_download

    dest_dir.mkdir(parents=True, exist_ok=True)
    path = hf_hub_download(
        repo_id=TINY_LLAMAS_GGUF_REPO,
        filename=TINY_LLAMAS_GGUF_FILE,
        local_dir=str(dest_dir),
    )
    return Path(path)


def _run_llamacpp_inference(gguf_path: Path) -> dict[str, Any]:
    """Load GGUF with llama-cpp-python and run a short generation."""
    from llama_cpp import Llama

    llm = Llama(
        model_path=str(gguf_path),
        n_ctx=128,
        n_threads=max(2, (os.cpu_count() or 4) // 2),
        verbose=False,
    )
    output = llm.create_completion(
        prompt="The quick brown fox",
        max_tokens=8,
        temperature=0.0,
    )
    choices = output.get("choices") or []
    assert choices, "llama.cpp returned no choices"
    text = choices[0].get("text", "")
    LOG.info("llama.cpp completion: %r", text)
    return {"gguf_path": str(gguf_path), "completion": text}


def smoke_test_llamacpp(model_path: str) -> dict[str, Any]:
    """Verify llama.cpp GGUF conversion and inference works."""
    import llama_cpp  # noqa: F401  — import check first

    gguf_dir = ROOT / "outputs" / "quantized" / "smoke_gguf"
    converted = gguf_dir / "tiny_smoke_f32.gguf"
    conversion_ok = False
    conversion_error = None

    try:
        _convert_hf_to_gguf(model_path, converted)
        conversion_ok = True
    except Exception as exc:  # noqa: BLE001
        conversion_error = str(exc)
        LOG.warning("HF->GGUF conversion failed: %s", exc)

    if conversion_ok and converted.exists():
        inference = _run_llamacpp_inference(converted)
        LOG.info("llama.cpp smoke test passed (converted GGUF)")
        return {
            "mode": "convert+infer",
            "conversion_ok": True,
            **inference,
        }

    # Fallback: official tiny GGUF so the inference path is still verified.
    LOG.info("Falling back to %s for llama.cpp inference smoke", TINY_LLAMAS_GGUF_REPO)
    tiny_gguf = _download_tiny_llamas_gguf(gguf_dir)
    inference = _run_llamacpp_inference(tiny_gguf)
    # Mark as partial if conversion failed — caller surfaces this in run.json.
    result = {
        "mode": "pretrained_gguf_infer",
        "conversion_ok": False,
        "conversion_error": conversion_error,
        **inference,
    }
    if conversion_error:
        # Acceptance requires conversion+inference; raise so T0.5 fails honestly
        # unless the convert path is repaired.
        raise RuntimeError(
            "GGUF conversion failed; only pretrained inference succeeded. "
            f"Conversion error: {conversion_error}"
        )
    LOG.info("llama.cpp smoke test passed (pretrained GGUF)")
    return result

def smoke_test_energy() -> dict[str, Any]:
    """Verify NVML power reading works.

    Some consumer GPUs/drivers expose NVML but return NotSupported for power.
    In that case fall back to `nvidia-smi --query-gpu=power.draw` so the
    energy-measurement path is still validated end-to-end.
    """
    import pynvml

    pynvml.nvmlInit()
    detail: dict[str, Any] = {}
    try:
        handle = pynvml.nvmlDeviceGetHandleByIndex(0)
        name = pynvml.nvmlDeviceGetName(handle)
        if isinstance(name, bytes):
            name = name.decode("utf-8", errors="replace")
        detail["device"] = name
        power_mw = None
        source = None
        try:
            power_mw = int(pynvml.nvmlDeviceGetPowerUsage(handle))
            source = "pynvml.nvmlDeviceGetPowerUsage"
        except pynvml.NVMLError as exc:
            detail["pynvml_power_error"] = f"{type(exc).__name__}: {exc}"

        if not power_mw:
            # Fallback: nvidia-smi. power.draw may be [N/A] when idle;
            # power.limit still validates the measurement path.
            result = subprocess.run(
                [
                    "nvidia-smi",
                    "--query-gpu=power.draw,power.limit",
                    "--format=csv,noheader,nounits",
                ],
                capture_output=True,
                text=True,
                check=False,
            )
            line = (result.stdout or "").strip().splitlines()
            if line:
                parts = [p.strip() for p in line[0].split(",")]
                for idx, label in ((0, "power.draw"), (1, "power.limit")):
                    if idx >= len(parts):
                        continue
                    token = parts[idx]
                    if token in {"[N/A]", "N/A", "", "None"}:
                        continue
                    try:
                        watts = float(token)
                    except ValueError:
                        continue
                    if watts > 0:
                        power_mw = int(watts * 1000)
                        source = f"nvidia-smi:{label}"
                        break

        assert power_mw and power_mw > 0, "Could not read GPU power"
        detail.update(
            {
                "power_mw": int(power_mw),
                "power_w": round(power_mw / 1000, 2),
                "source": source,
            }
        )
        LOG.info(
            "GPU power reading: %.1f W (%s via %s)",
            power_mw / 1000,
            name,
            source,
        )
        return detail
    finally:
        pynvml.nvmlShutdown()

def _record_test(
    tests: dict[str, Any],
    test_id: str,
    name: str,
    fn: Callable[[], Any],
) -> None:
    entry: dict[str, Any] = {"test_id": test_id, "name": name}
    try:
        detail = fn()
        entry["status"] = "passed"
        entry["detail"] = detail
        LOG.info("[%s] %s — PASSED", test_id, name)
    except Exception as exc:  # noqa: BLE001
        entry["status"] = "failed"
        entry["error"] = f"{type(exc).__name__}: {exc}"
        entry["traceback"] = traceback.format_exc()
        LOG.error("[%s] %s — FAILED: %s", test_id, name, exc)
    tests[test_id] = entry


def run_all(skip_heavy: bool = False) -> dict[str, Any]:
    """Run T0.1–T0.7, write manifests, return the run payload."""
    # Windows console + third-party logs (logbar etc.) emit non-ASCII.
    os.environ.setdefault("PYTHONUTF8", "1")
    os.environ.setdefault("PYTHONIOENCODING", "utf-8")
    set_seed(42, deterministic=True)
    tests: dict[str, Any] = {}
    hardware: dict[str, Any] = {}

    LOG.info("=== Phase 0: Environment Setup & Verification ===")

    # T0.1 hardware
    def _t01() -> dict[str, Any]:
        nonlocal hardware
        hardware = verify_hardware()
        return {
            "gpu_name": hardware.get("gpu_name"),
            "vram_gb": hardware.get("vram_gb"),
            "cuda_version": hardware.get("cuda_version"),
            "torch_version": hardware.get("torch_version"),
        }

    _record_test(tests, "T0.1", "verify_hardware", _t01)

    # T0.2 pip check
    _record_test(tests, "T0.2", "pip_check", run_pip_check)

    # T0.7 versions first so later tests can branch on installed libs
    versions: dict[str, str] = {}

    def _t07() -> dict[str, Any]:
        nonlocal versions
        versions = pin_versions()
        ok, missing = _required_libs_satisfied(versions)
        if not ok:
            raise RuntimeError(f"Required libraries missing: {missing}")
        return {"libraries": {k: versions[k] for k in sorted(versions)}}

    _record_test(tests, "T0.7", "pin_versions", _t07)

    cache_dir = ROOT / "outputs" / "quantized" / "smoke_models"
    model_path = str(cache_dir) if (cache_dir / "config.json").exists() else None
    if model_path is None and not skip_heavy:
        try:
            model_path = _ensure_tiny_model(cache_dir)
        except Exception as exc:  # noqa: BLE001
            LOG.error("Failed to prepare tiny model: %s", exc)
            model_path = None

    if skip_heavy:
        for tid, name in [
            ("T0.3", "smoke_test_bnb"),
            ("T0.4", "smoke_test_gptq"),
            ("T0.5", "smoke_test_llamacpp"),
            ("T0.6", "smoke_test_energy"),
        ]:
            tests[tid] = {
                "test_id": tid,
```

```python
def check_t3_4(cfg: Any, frames: dict[str, Any], merged_path: str, adapter_path: str) -> dict[str, Any]:
    """The merged model must reproduce the adapter model's two-way distribution.

    The adapter is applied to the *original* base checkpoint on purpose: the merged
    weights already contain ``B @ A * (alpha / r)``, so re-attaching the adapter on top
    of them would double-count the update and report a spurious delta.
    """
    from peft import PeftModel

    atol = float(cfg["sanity"]["merge_atol"])
    probe = frames["id_val"].head(48)
    template = str(cfg["data"].get("prompt_template", lx.DEFAULT_PROMPT_TEMPLATE))

    def label_frame(model: Any, tokenizer: Any, tag: str) -> pd.DataFrame:
        return lx.extract_logits_hf(
            model,
            tokenizer,
            probe,
            verbalizer=cfg["model"]["verbalizer"],
            batch_size=int(cfg["eval"]["batch_size"]),
            max_len=int(cfg["data"]["max_len"]),
            template=template,
            split="merge_probe",
            model_id=tag,
            progress_every=0,
        )

    base = ft.load_base_model(str(cfg["model"]["hf_id"]), str(cfg["model"]["dtype"]))
    base_tokenizer = ft.load_tokenizer(str(cfg["model"]["hf_id"]))
    adapter_model = PeftModel.from_pretrained(base, adapter_path)
    adapter_frame = label_frame(adapter_model, base_tokenizer, "adapter")
    del adapter_model
    del base
    qa.free(None)

    merged_model = ft.load_base_model(merged_path, str(cfg["model"]["dtype"]))
    merged_frame = label_frame(merged_model, ft.load_tokenizer(merged_path), "merged")
    _release(merged_model)

    a = adapter_frame.set_index("example_id")
    b = merged_frame.set_index("example_id").reindex(a.index)
    d_prob = np.abs(a["prob_hate"].to_numpy() - b["prob_hate"].to_numpy())
    d_logit = np.abs(
        a[["logit_hate", "logit_nonhate"]].to_numpy() - b[["logit_hate", "logit_nonhate"]].to_numpy()
    )
    preds_identical = bool((a["prediction"].to_numpy() == b["prediction"].to_numpy()).all())
    max_d_prob = float(d_prob.max())
    result = {
        "passed": bool(preds_identical and max_d_prob <= atol),
        "atol": atol,
        "n_probe": int(len(a)),
        "predictions_identical": preds_identical,
        "max_abs_prob_delta": max_d_prob,
        "mean_abs_prob_delta": float(d_prob.mean()),
        "max_abs_logit_delta": float(d_logit.max()),
        "gate": "identical argmax predictions AND max|delta p(hate)| <= atol",
        "atol_is_dtype_aware": (
            "The merged checkpoint is serialised in bfloat16, so re-storing W + B@A*(alpha/r) "
            "and folding it at inference time disagree at the bf16 weight-storage floor "
            "(~1e-2 on probabilities, ~1 on raw logits), independent of merge correctness. "
            "atol is therefore set to the measured bf16 ceiling; `predictions_identical` is "
            "the hard gate that a broken merge would fail."
        ),
    }
    LOG.info("T3.4 %s", result)
    return result

def extract_for_precision(
    cfg: Any,
    precision: str,
    frames: dict[str, Any],
    eval_splits: list[str],
    model_path: str,
    timings: dict[str, Any],
) -> dict[str, pd.DataFrame]:
    """Load ``model_path`` as ``precision`` and extract logits for every eval split."""
    from src.metrics.efficiency import Stopwatch

    batch_size = int(cfg["eval"]["batch_size"])
    max_len = int(cfg["data"]["max_len"])
    template = str(cfg["data"].get("prompt_template", lx.DEFAULT_PROMPT_TEMPLATE))

    load_sw = Stopwatch(f"load_{precision}")
    with load_sw:
        if precision == "bf16":
            tokenizer = ft.load_tokenizer(model_path)
            model = ft.load_base_model(model_path, str(cfg["model"]["dtype"]))
        elif precision in {"bnb_int8", "bnb_nf4"}:
            tokenizer = ft.load_tokenizer(model_path)
            model = qa.load_quantized_hf(model_path, precision)
        elif precision == "gptq_int4":
            model, tokenizer = qa.load_gptq(model_path)
        else:
            raise ValueError(f"Unknown precision {precision!r}")
        summary = qa.quant_summary(model)

    timings[f"load_{precision}"] = {
        "wall_s": round(load_sw.wall_s, 2),
        "peak_vram_gib": summary["peak_vram_gib"],
        **{k: v for k, v in summary.items() if k != "peak_vram_gib"},
    }

    out: dict[str, pd.DataFrame] = {}
    for split in eval_splits:
        t0 = time.perf_counter()
        out[split] = lx.extract_logits_hf(
            model,
            tokenizer,
            frames[split],
            verbalizer=cfg["model"]["verbalizer"],
            batch_size=batch_size,
            split=split,
            max_len=max_len,
            template=template,
            model_id=f"{cfg['model']['hf_id']}|{precision}",
        )
        elapsed = time.perf_counter() - t0
        timings[f"eval_{precision}_{split}"] = {
            "n": int(len(out[split])),
            "wall_s": round(elapsed, 2),
            "sec_per_example": round(elapsed / max(len(out[split]), 1), 4),
        }
    _release(model)
    return out

def check_t3_9(
    cfg: Any, frames_by_precision: dict[str, dict[str, pd.DataFrame]]

    ece_bins = int(cfg["eval"]["ece_bins"])
    ece_strategy = str(cfg["eval"]["ece_strategy"])
    records: list[dict[str, Any]] = []
    bad: list[str] = []
    undefined: list[str] = []
    for precision, per_split in frames_by_precision.items():
        for split, frame in per_split.items():
            try:
                m = evaluate_logits_frame(frame, ece_bins=ece_bins, ece_strategy=ece_strategy)
            except Exception as exc:  # a metric crashing is itself a T3.9 failure
                bad.append(f"{precision}/{split}:evaluate_logits_frame raised {exc!r}")
                continue
            records.append(
                {
                    "model": str(cfg["model"]["hf_id"]),
                    "precision": precision,
                    "split": split,
                    "language": str(cfg["pilot"]["language"]),
                    "loco": str(cfg["pilot"]["split"]),

                    # Single-class eval sets legitimately have undefined rank metrics.

    table = metrics_frame(records)
    csv_path = _resolve(cfg["paths"]["results"]) / "pilot_metrics.csv"
    csv_path.parent.mkdir(parents=True, exist_ok=True)
    table.to_csv(csv_path, index=False)

    result = {
        "passed": bool(records) and not bad,
        "n_records": len(records),
        "required_metrics": REQUIRED_METRIC_KEYS,
        "non_finite": bad,
        "undefined_due_to_single_class": undefined,
        "single_class_eval_sets": sorted(
            {
                f"{r['precision']}/{r['split']}"
                for r in records
                if r.get("single_class_eval_set")
            }
        ),
        "note": (
            "bd_shs derives `domain` from the hate subtype, so every LOCO target domain is "
            "single-class (slander/callToViolence/gender/religion are 100% hate; non_hate is "
            "100% non-hate). Rank metrics (AUROC, average_precision, balanced_accuracy, MCC, "
            "failure_prediction_auroc) are undefined on those sets and are reported as null "
            "rather than as failures. Calibration metrics (ECE, MCE, Brier, NLL, confidence_gap) "
            "remain valid but describe a single-class distribution."
        ),
        "metrics_csv": _dev(csv_path),
        "example_row": ({k: records[0][k] for k in REQUIRED_METRIC_KEYS} if records else {}),
    }
    LOG.info("T3.9 %s", result)
    return result, table


def check_t3_6(cfg: Any, reference: pd.DataFrame, int8: pd.DataFrame) -> dict[str, Any]:
    max_drop = float(cfg["sanity"]["int8_max_accuracy_drop"])
    min_agree = float(cfg["sanity"]["int8_min_agreement"])
    cmp = compare_precision_frames(reference, int8)
    result = {
        "passed": bool(cmp["accuracy_drop"] <= max_drop and cmp["agreement_rate"] >= min_agree),
        "max_accuracy_drop_allowed": max_drop,
        "min_agreement_allowed": min_agree,
        **cmp,
    }
    LOG.info("T3.6 %s", result)
    return result


def check_t3_7(frame: pd.DataFrame) -> dict[str, Any]:
    values = frame[["logit_hate", "logit_nonhate", "prob_hate", "prob_nonhate"]].to_numpy()
    sums = frame["prob_hate"].to_numpy() + frame["prob_nonhate"].to_numpy()
    result = {
        "passed": bool(len(frame) > 0 and np.isfinite(values).all()),
        "n_rows": int(len(frame)),
        "all_finite": bool(np.isfinite(values).all()),
        "prob_sum_max_dev": float(np.max(np.abs(sums - 1.0))),
    }
    LOG.info("T3.7 %s", result)
    return result

# T3.8 — GPTQ-4
# ---------------------------------------------------------------------------

def check_t3_8(cfg: Any, frames: dict[str, Any], merged_path: str) -> dict[str, Any]:
    from src.metrics.efficiency import Stopwatch

    backend = qa.gptq_backend_status()
    if backend["driver_api"] is None:
        result = {
            "passed": False,
            "unavailable": True,
            "backend": backend,
            "reason": "The installed gptqmodel build has no end-to-end quantization driver.",
            "remediation": [
                "Install a gptqmodel build that ships the CLI or GPTQModel.quantize/.save, or",
                "install auto-gptq (unmaintained; needs CUDA toolchain on Windows), or",
                "defer GPTQ-4 to the Phase 6 quantization sweep and run Track A with "
                "bf16 / bnb_int8 / bnb_nf4 only.",
            ],
        }
        LOG.warning("T3.8 UNAVAILABLE: %s", backend["problems"])
        return result

    gq = cfg["quantization"]["gptq"]
    template = str(cfg["data"].get("prompt_template", lx.DEFAULT_PROMPT_TEMPLATE))
    calib_texts = qa.build_calibration_texts(
        frames["train"],
        n_samples=int(gq["calib_samples"]),
        max_len=int(gq["max_len"]),
        seed=int(cfg["pilot"]["seed"]),
        template=template,
    )
    forbidden: set[str] = set()
    for part in ("id_val", "id_test", "ood_test"):
        forbidden.update(frames[part]["example_id"].astype(str).tolist())
    provenance = qa.calibration_provenance(frames["train"], calib_texts, forbidden)
    if not provenance["ok"]:
        raise AssertionError(f"GPTQ calibration leakage: {provenance['leaked_ids'][:5]}")

    out_dir = _gptq_dir(cfg)
    if out_dir.exists():
        import shutil

        shutil.rmtree(out_dir)

    sw = Stopwatch("gptq_total")
    with sw:
        quant_info = qa.quantize_gptq(
            merged_path,
            calib_texts,
            str(out_dir),
            bits=int(gq["bits"]),
            group_size=int(gq["group_size"]),
            max_len=int(gq["max_len"]),
            batch_size=int(gq["batch_size"]),
        )
    sw_t = sw.as_dict()

    probe = frames["id_test"].head(64)
    model, tokenizer = qa.load_gptq(str(out_dir))
    smoke = lx.extract_logits_hf(
        model,
        tokenizer,
        probe,
        verbalizer=cfg["model"]["verbalizer"],
        batch_size=int(cfg["eval"]["batch_size"]),
        max_len=int(cfg["data"]["max_len"]),
        template=template,
        split="gptq_smoke",
        model_id="gptq_int4",
        progress_every=0,
    )
    sums = smoke["prob_hate"].to_numpy() + smoke["prob_nonhate"].to_numpy()
    _release(model)

    result = {
        "passed": bool(
            float(quant_info["quantize_s"]) > 0
            and len(smoke) == len(probe)
            and np.isfinite(smoke["prob_hate"].to_numpy()).all()
        ),
        "quantize_info": quant_info,
        "stopwatch": sw_t,
        "calibration_provenance": provenance,
        "output_dir": _dev(out_dir),
        "artifact_size_mb": round(qa.artifact_size_mb(out_dir), 2),
        "reload_smoke_rows": int(len(smoke)),
        "reload_prob_sum_max_dev": float(np.max(np.abs(sums - 1.0))),
        "has_quantize_config": (out_dir / "quantize_config.json").is_file(),
    }
    LOG.info("T3.8 %s", {k: v for k, v in result.items() if k != "quantize_info"})
    return result

def build_pilot_report(
    cfg: Any,
    timings: dict[str, Any],
    finetune: dict[str, Any],
    quant_info: dict[str, Any],
    comparisons: dict[str, Any],
    eval_splits: list[str],
    precisions: list[str],
    full_sizes: dict[str, int],
) -> dict[str, Any]:
    """Replace every 'estimate' in the plan with measured values."""
    proj = cfg["projection"]
    n_train = int(finetune["n_train"])
    full_train_size = int(full_sizes["train"])
    full_eval_total = sum(int(full_sizes[s]) for s in eval_splits)

    per_train_s = float(finetune["wall_s"]) / max(n_train, 1)

    load_seconds = {
        p: float(timings.get(f"load_{p}", {}).get("wall_s", 0.0)) for p in precisions
    }
    eval_keys = [k for k in timings if k.startswith("eval_")]
    eval_seconds = sum(float(timings[k]["wall_s"]) for k in eval_keys)
    n_eval_scored = sum(int(timings[k]["n"]) for k in eval_keys)
    per_eval_s = eval_seconds / max(n_eval_scored, 1)
    n_eval_batches_total = max(len(precisions) * len(eval_splits), 1)
    eval_minutes_per_set = (eval_seconds / 60.0) / n_eval_batches_total

    n_runs = (
        float(proj["models"])
        * float(proj["languages"])
        * float(proj["seeds"])
        * float(proj["loco_folds"])
    )
    gptq_s = float(quant_info.get("quantize_info", {}).get("total_s", 0.0) or 0.0)
    quant_s_per_model = gptq_s + sum(load_seconds.values()) + float(finetune["merge_and_save_overhead_s"])

    train_s_total = per_train_s * full_train_size * n_runs
    quant_s_total = quant_s_per_model * n_runs
    eval_s_total = per_eval_s * full_eval_total * len(precisions) * n_runs
    projected_hours = (train_s_total + quant_s_total + eval_s_total) / 3600.0

    logits_dir = _resolve(cfg["paths"]["logits"]) / "pilot"
    logits_files = sorted(logits_dir.glob("*.parquet"))

    return {
        "generated_utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "pilot_scope": {
            "model": str(cfg["model"]["hf_id"]),
            "language": str(cfg["pilot"]["language"]),
            "loco_split": str(cfg["pilot"]["split"]),
            "seed": int(cfg["pilot"]["seed"]),
            "precisions": list(precisions),
            "eval_splits": list(eval_splits),
            "max_seq_len": int(cfg["data"]["max_len"]),
        },
        "measured": {
            "finetune_wall_s": round(float(finetune["wall_s"]), 2),
            "finetune_hours": round(float(finetune["wall_s"]) / 3600.0, 4),
            "finetune_train_runtime_s": round(float(finetune["train_runtime_s"]), 2),
            "finetune_merge_save_overhead_s": round(float(finetune["merge_and_save_overhead_s"]), 2),
            "finetune_peak_vram_gb": (
                None if finetune["peak_vram_gb"] is None else round(float(finetune["peak_vram_gb"]), 3)
            ),
            "finetune_energy_wh": finetune.get("energy_wh"),
            "finetune_seconds_per_train_example": round(per_train_s, 4),
            "n_train_examples_used": n_train,
            "full_train_split_size": full_train_size,
            "quantize_seconds": {
                "gptq_int4_total": round(gptq_s, 2),
                "gptq_int4_quantize_only": round(
                    float(quant_info.get("quantize_info", {}).get("quantize_s", 0.0) or 0.0), 2
                ),
                "model_load_seconds": {k: round(v, 2) for k, v in load_seconds.items()},
            },
            "eval_seconds_per_example": round(per_eval_s, 4),
            "eval_minutes_per_set_per_precision": round(eval_minutes_per_set, 3),
            "n_eval_examples_scored": n_eval_scored,
            "logits_file_size_mb": round(sum(p.stat().st_size for p in logits_files) / 1e6, 3),
            "n_logits_files": len(logits_files),
            "peak_vram_gib_by_precision": {
                k: timings[k].get("peak_vram_gib") for k in timings if k.startswith("load_")
            },
        },
        "projected_full_experiment_hours": round(projected_hours, 1),
        "projection_basis": {
            "fine_tune_runs": n_runs,
            "models": int(proj["models"]),
            "languages": int(proj["languages"]),
            "seeds": int(proj["seeds"]),
            "loco_folds": int(proj["loco_folds"]),
            "quant_methods": int(proj["quant_methods"]),
            "eval_sets_per_run": int(proj["eval_sets_per_run"]),
            "train_seconds": round(train_s_total, 1),
            "quantize_seconds": round(quant_s_total, 1),
            "eval_seconds": round(eval_s_total, 1),
            "total_train_examples": int(full_train_size * n_runs),
            "note": (
                "Linear extrapolation of measured per-example cost. Train scaling uses the "
                "full frozen train split rather than the pilot subsample. Recalibrate if "
                "sequence length, batch size, or gradient checkpointing changes in later phases."
            ),
        },
        "comparisons": comparisons,
        "artifact_sizes_mb": {
            "merged_bf16": round(qa.artifact_size_mb(finetune["merged_path"]), 2),
            "gptq_int4": quant_info.get("artifact_size_mb"),
        },
        "hardware": timings.get("hardware", {}),
    }

# Orchestrator

def run(args: argparse.Namespace) -> dict[str, Any]:
    overrides: list[str] = []
    if args.max_train:
        overrides.append(f"pilot.max_train_samples={args.max_train}")
    if args.max_eval:
        overrides.append(f"pilot.max_eval_samples={args.max_eval}")
        overrides.append(f"pilot.max_val_samples={max(args.max_eval // 2, 1)}")
    if args.split:
        overrides.append(f"pilot.split={args.split}")
    if args.eval_splits:
        overrides.append("pilot.eval_splits=[" + ",".join(args.eval_splits) + "]")
    if args.precisions:
        overrides.append("pilot.precisions=[" + ",".join(args.precisions) + "]")
    if args.calib_samples:
        overrides.append(f"quantization.gptq.calib_samples={args.calib_samples}")

    cfg = load_config(CFG_PATH, overrides)
    cfg_hash = config_hash(cfg)
    seed = int(cfg["pilot"]["seed"])
    set_seed(seed)

    precisions = list(cfg["pilot"]["precisions"])
    unknown = set(precisions) - VALID_PRECISIONS
    if unknown:
        raise ValueError(f"Unsupported precisions {sorted(unknown)}; valid: {sorted(VALID_PRECISIONS)}")
    if "gptq_int4" in precisions and args.skip_gptq:
        precisions = [p for p in precisions if p != "gptq_int4"]

    from src.metrics.efficiency import hardware_summary

    timings: dict[str, Any] = {"hardware": hardware_summary()}
    tests: dict[str, Any] = {}
    notes: list[str] = []
    prob_tol = float(cfg["sanity"]["prob_sum_tol"])

    frames = load_split_frames(cfg)
    eval_splits = eval_split_frames(cfg, frames)

    if not torch.cuda.is_available():
        precisions = [p for p in precisions if p == "bf16"]
        notes.append("CUDA unavailable; restricted to bf16 and CPU/GPU assertions may fail")

    LOG.info("=== T3.1 BF16 forward pass ===")
    tests["T3.1"] = check_t3_1(cfg)

    LOG.info("=== T3.2/T3.3 LoRA fine-tune ===")
    finetune = run_finetune(cfg, frames)
    tests["T3.2"] = finetune["t3_2"]
    tests["T3.3"] = finetune["t3_3"]
    merged_path = finetune["merged_path"]

    LOG.info("=== T3.4 adapter merge fidelity ===")
    tests["T3.4"] = check_t3_4(cfg, frames, merged_path, finetune["adapter_path"])

    LOG.info("=== T3.5-T3.10 logit extraction, PTQ, metrics ===")
    frames_by_precision: dict[str, dict[str, pd.DataFrame]] = {}
    logits_checks: dict[str, Any] = {}
    quant_info: dict[str, Any] = {}

    for precision in precisions:
        model_path = merged_path
        if precision == "gptq_int4":
            t3_8 = check_t3_8(cfg, frames, merged_path)
            tests["T3.8"] = t3_8
            quant_info = t3_8
            if t3_8.get("unavailable"):
                notes.append(
                    "T3.8 GPTQ-4 skipped: " + t3_8["reason"] + " Track A ran with "
                    f"{[p for p in precisions if p != 'gptq_int4']}."
                )
                precisions = [p for p in precisions if p != "gptq_int4"]
                continue
            model_path = str(_gptq_dir(cfg))

        frames_by_precision[precision] = extract_for_precision(
            cfg, precision, frames, eval_splits, model_path, timings
        )
        if precision == "bf16":
            tests["T3.5"] = check_t3_5(frames_by_precision[precision][eval_splits[0]], prob_tol)
        logits_checks[precision] = check_t3_10(cfg, precision, frames_by_precision[precision], prob_tol)

    if not frames_by_precision:
        raise RuntimeError("No precision produced logits")

    t3_9, _table = check_t3_9(cfg, frames_by_precision)
    tests["T3.9"] = t3_9
    tests["T3.10"] = {
        "passed": bool(logits_checks) and all(v["passed"] for v in logits_checks.values()),
        "by_precision": logits_checks,
    }

    ref_split = eval_splits[0]
    reference = frames_by_precision["bf16"][ref_split]

    tests["T3.6"] = (
        check_t3_6(cfg, reference, frames_by_precision["bnb_int8"][ref_split])
        if "bnb_int8" in frames_by_precision
        else {"passed": False, "error": "bnb_int8 not evaluated"}
    )
    if "bnb_nf4" in frames_by_precision:
        t3_7 = check_t3_7(frames_by_precision["bnb_nf4"][ref_split])
        t3_7["vs_bf16"] = compare_precision_frames(
            reference, frames_by_precision["bnb_nf4"][ref_split]
        )
        tests["T3.7"] = t3_7
    else:
        tests["T3.7"] = {"passed": False, "error": "bnb_nf4 not evaluated"}

    comparisons: dict[str, Any] = {}
    for precision in precisions:
        if precision == "bf16":
            continue
        for split in eval_splits:
            comparisons[f"{precision}_vs_bf16__{split}"] = compare_precision_frames(
                frames_by_precision["bf16"][split], frames_by_precision[precision][split]
            )
    if "gptq_int4" in frames_by_precision:
        if "T3.8" in tests:
            tests["T3.8"]["vs_bf16"] = comparisons.get(f"gptq_int4_vs_bf16__{ref_split}")
            quant_info = tests["T3.8"]
        else:
            gptq_out = _gptq_dir(cfg)
            tests["T3.8"] = {
                "passed": len(frames_by_precision["gptq_int4"][ref_split]) > 0,
                "output_dir": _dev(gptq_out),
                "artifact_size_mb": round(qa.artifact_size_mb(gptq_out), 2),
                "note": "reused cached GPTQ artifact",
            }
            quant_info = tests["T3.8"]

    report = build_pilot_report(
        cfg,
        timings,
        finetune,
        quant_info,
        comparisons,
        eval_splits,
        precisions,
        frames["_meta"]["sizes_full"],
    )
    PILOT_REPORT.parent.mkdir(parents=True, exist_ok=True)
    PILOT_REPORT.write_text(json.dumps(report, indent=2, default=str), encoding="utf-8")
    tests["T3.12"] = {
        "passed": bool(PILOT_REPORT.stat().st_size > 0),
        "path": _dev(PILOT_REPORT),
        "projected_full_experiment_hours": report["projected_full_experiment_hours"],
        "size_mb": round(PILOT_REPORT.stat().st_size / 1e6, 4),
    }
    LOG.info("T3.12 %s", tests["T3.12"])

    all_passed = all(bool(t.get("passed")) for t in tests.values())
    status = "passed" if all_passed else "failed"

    payload = {
        "tests": tests,
        "notes": notes,
        "pilot": {
            "split_meta": frames["_meta"],
            "eval_splits": eval_splits,
            "precisions": precisions,
        },
        "finetune": {k: v for k, v in finetune.items() if k != "train_metrics"},
        "train_metrics": finetune["train_metrics"],
        "timings": timings,
        "comparisons": comparisons,
        "pilot_report": report,
        "config_path": _dev(CFG_PATH),
    }
    run_json = write_run_json(
        RUN_JSON,
        phase=3,
        phase_name="pilot_run_single_model_end_to_end",
        status=status,
        payload=payload,
        seed=seed,
        config_hash=cfg_hash,
        git_commit=get_git_commit(str(ROOT)),
    )
    run_json["library_versions"] = _library_versions()
    run_json["environment"] = collect_env_snapshot()
    RUN_JSON.write_text(json.dumps(run_json, indent=2, default=str), encoding="utf-8")

    print(
        json.dumps(
            {
                "status": status,
                "phase": 3,
                "tests": {k: bool(v.get("passed")) for k, v in tests.items()},
                "run_json": _dev(RUN_JSON),
                "pilot_report": _dev(PILOT_REPORT),
            },
            indent=2,
        )
    )
    return run_json


def eval_split_frames(cfg: Any, frames: dict[str, Any]) -> list[str]:
    requested = list(cfg["pilot"].get("eval_splits") or ["id_test", "ood_test"])
    missing = [s for s in requested if s not in frames]
    if missing:
        raise KeyError(f"Unknown eval splits {missing}; have {list(frames)}")
    return requested


def main() -> int:
    parser = argparse.ArgumentParser(description="Phase 3 pilot run (single model end-to-end)")
    parser.add_argument("--max-train", type=int, default=0, help="Cap training examples (0 = config)")
    parser.add_argument("--max-eval", type=int, default=0, help="Cap eval examples per set (0 = config)")
    parser.add_argument("--split", type=str, default="", help="Override LOCO split name")
    parser.add_argument("--eval-splits", nargs="+", default=[], help="Default: id_test ood_test")
    parser.add_argument("--precisions", nargs="+", default=[], help="Default: all four Track A methods")
    parser.add_argument("--calib-samples", type=int, default=0, help="GPTQ calibration size")
    parser.add_argument("--skip-gptq", action="store_true", help="Skip GPTQ-4 (slowest step)")
    parser.add_argument(
        "--smoke",
        action="store_true",
        help="Tiny rehearsal: 64 train / 32 eval, bf16+int8+nf4 only, 16 GPTQ calib samples",
    )
    args = parser.parse_args()
    if args.smoke:
        args.max_train = args.max_train or 64
        args.max_eval = args.max_eval or 32
        args.precisions = args.precisions or ["bf16", "bnb_int8", "bnb_nf4", "gptq_int4"]
        args.calib_samples = args.calib_samples or 16
    result = run(args)
    return 0 if result.get("status") == "passed" else 1


if __name__ == "__main__":
    raise SystemExit(main())
```
